290inline Order* OrderBook::allocateOrder() {
291 auto *order = order_buffer_.allocate();
292
293 // The generator seeds a Mersenne twister from the OS entropy source when it is
294 // constructed, so building one per order - as `random_generator()()` did - cost
295 // far more than the id it produced. One per thread instead; orders are
296 // allocated on the exchange thread, and thread_local keeps that true if that
297 // ever stops holding.
298 static thread_local boost::uuids::random_generator uuid_gen;
299 const boost::uuids::uuid u = uuid_gen();
300
301 // A UUID renders as exactly 36 characters, which is the capacity of the id
302 // field, so it goes straight into the order's own buffer. lexical_cast built a
303 // std::string first and threw it away.
304 static_assert(decltype(Order::order_id)::capacity() == 36,
305 "order_id must hold a rendered UUID exactly");
306 boost::uuids::to_chars(u, order->order_id.data());
307 order->order_id.data()[36] = '\0';
308
309 order->id = utils::nextOrderId();
310 order->created_time = utils::get_current_time_ns();
311 order->last_update_time = order->created_time;
312 order->resetFillAccounting();
313 return order;
314}

332inline void OrderBook::onOrderUpdate(const OrderUpdate& update) {
333 auto *our_order = findOrder(update.order_id);
334 if (!our_order) {
335 // This is a MD order, update md_level_qty
336 if (update.price != update.old_price) {
337 auto it = feed_md_level_quantity_.find(update.old_price);
338 if (it != feed_md_level_quantity_.end()) {
339 it->second -= update.old_qty;
340 }
341
342 it = feed_md_level_quantity_.find(update.price);
343 if (it != feed_md_level_quantity_.end()) {
344 it->second += update.quantity;
345 } else if (update.quantity) {
346 feed_md_level_quantity_[update.price] = update.quantity;
347 }
348 }
349 else {
350 auto it = feed_md_level_quantity_.find(update.price);
351 if (it != feed_md_level_quantity_.end()) {
352 it->second += update.quantity - update.old_qty;
353 }
354 else if (update.quantity) {
355 feed_md_level_quantity_[update.price] = update.quantity;
356 }
357 }
358 }
359 else if (update.quantity == 0 && our_order->quantity != update.quantity) {
360 our_order->leaves_quantity = 0;
361 }
362}

393 std::stringstream ss;
394 ss << "OrderBook: " << symbol_ << "\n";
395 auto &sell_levels = getLevelsL3(orderbook::Side::Sell);
396 auto it_sell = sell_levels.rend();
397 std::advance(it_sell, -20);
398 for (; it_sell != sell_levels.rend(); ++it_sell) {
399 ss << std::right << std::setw(50) << std::format("{:.2f}", to_price_double(it_sell->first));
400 ss << std::right << std::setw(20) << std::format("{:.8f}", to_qty_double(it_sell->second.total_quantity));
401 ss << std::right << std::setw(10) << it_sell->second.orderCount() << "\n";
402 }
403
404 auto &buy_levels = getLevelsL3(orderbook::Side::Buy);
405 uint32_t i = 0;
406 for (auto it_buy = buy_levels.begin(); it_buy != buy_levels.end() && i < 20; ++it_buy, ++i) {
407 ss << std::right << std::setw(10) << it_buy->second.orderCount();
408 ss << std::right << std::setw(20) << std::format("{:.8f}", to_qty_double(it_buy->second.total_quantity));
409 ss << std::right << std::setw(20) << std::format("{:.2f}", to_price_double(it_buy->first)) << "\n";
410 }
411 return ss.str();
412}

437 auto &ask_levels = getLevelsL3(orderbook::Side::Sell);
438 auto &bid_levels = getLevelsL3(orderbook::Side::Buy);
439 auto sz = sizeof(MarketDataUpdate) + sizeof(MDSubscriptionResponse) + sizeof(BookSnapshot) + (bid_levels.size() + ask_levels.size()) * sizeof(MDLevel);
440 auto index = md_update_queue.reserve(static_cast<uint32_t>(sz));
441 auto *update = reinterpret_cast<MarketDataUpdate*>(md_update_queue[index]);
442 memcpy(update->symbol, symbol_.data(), sizeof(update->symbol));
443 update->venue = venue_;
444 update->type = MDUpdateType::SUB_RESPONSE;
445
446 auto *response = reinterpret_cast<MDSubscriptionResponse*>(update->data);
447 response->channel = channel;
448
449 auto *snapshot = reinterpret_cast<BookSnapshot*>(response->data);
450 snapshot->num_bid = static_cast<uint32_t>(bid_levels.size());
451 snapshot->num_ask = static_cast<uint32_t>(ask_levels.size());
452
453 auto *levels = reinterpret_cast<MDLevel*>(snapshot->levels);
454
455 for (auto it = bid_levels.begin(); it != bid_levels.end(); ++it, ++levels) {
456 levels->price = it->first;
457 levels->qty = it->second.total_quantity;
458 levels->num_orders = static_cast<uint32_t>(it->second.orderCount());

460 for (auto it = ask_levels.begin(); it != ask_levels.end(); ++it, ++levels) {

462 levels->qty = it->second.total_quantity;
