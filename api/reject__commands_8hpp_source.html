18bool toggle_id(Set &ids, const Command &command, std::string &out) {
19 const auto id = command.arg_view();
20 if (id.empty()) {
21 out = std::format("{} needs an ID", to_string(command.command));
22 return false;
23 }
24 const bool on = command.flag != 0;
25 bool changed;
