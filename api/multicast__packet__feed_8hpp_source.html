114 bool add_line(const char *key, uint8_t line) {
115 // The receiver thread walks lines_, so a line added under it would be a data race.
116 if (started_ || !config_.contains(key) || line >= MAX_LINES) {
117 return false;
118 }
119 const auto &group = config_[key];
120 slick::socket::PolledMulticastReceiverConfig receiver;
121 receiver.multicast_address = group.at("address").get<std::string>();
122 receiver.port = group.at("port").get<uint16_t>();
123 receiver.interface_address = config_.value("interface", receiver.interface_address);
124 receiver.receive_buffer_size = config_.value("receive_buffer_size", 1 << 26);
125 if (receivers_[line]) {
126 return false; // the line is taken
127 }
128 receivers_[line] = std::make_unique<slick::socket::PolledMulticastReceiver>(
129 std::format("{} {}", name_, key), receiver);
130 lines_.push_back(line);
131 return true;
132 }

134 void start() {
135 if (thread_.joinable()) {
136 LOG_WARN("feed {}: already started", name_); // as slick-socket's own receivers do
137 return;
138 }
139 started_ = true;
140 for (const uint8_t line : lines_) {
141 if (!receivers_[line]->open()) {
142 LOG_ERROR("feed {}: a multicast receiver failed to start", name_);
143 }
144 }
145 // Starting joined every group. A line the venue still wants left stays
146 // muted, and its next set_joined() leaves the group again.
147 left_ = 0;
148 failing_ = 0;
149 retry_at_ns_.fill(0);
150 for (auto &request : request_) {
151 request.store(NO_REQUEST, std::memory_order_relaxed);
152 }
153 pending_op_.fill(NO_REQUEST);
154 if (!lines_.empty()) {
155 running_.store(true, std::memory_order_release);
156 thread_ = std::thread([this] { receive_loop(); });
157 }
158 }

160 void stop() {
161 running_.store(false, std::memory_order_release);
162 if (thread_.joinable()) {
163 thread_.join();
164 }
165 for (const uint8_t line : lines_) {
166 receivers_[line]->close();
167 }
168 started_ = false;

178 bool deliver(uint8_t line, const uint8_t *data, size_t size) {
179 assert(line < MAX_LINES);
180 if (muted_.load(std::memory_order_relaxed) & (1u << line)) {
181 return false;
182 }
183 const auto n = static_cast<uint32_t>(size + 1);
184 if (n > packets_.size()) [[unlikely]] {
185 // A queue too small for one packet is too small for all of them: log the
186 // first, count the rest, rather than flood the log at the feed's rate.
187 if (oversized_.fetch_add(1, std::memory_order_relaxed) == 0) {
188 LOG_ERROR("feed {}: line {} packet of {} bytes exceeds the {} byte queue; dropping it and any like it",
189 name_, line, size, packets_.size());
190 }
191 return false;
192 }
193 const uint64_t index = packets_.reserve(n);
194 uint8_t *entry = packets_[index];
195 entry[0] = line;
196 std::memcpy(entry + 1, data, size);
197 packets_.publish(index, n);
198 return true;
199 }

222 void set_joined(uint8_t line, bool joined, uint64_t now_ns) {
223 assert(line < MAX_LINES);
224 const uint32_t bit = 1u << line;
225 const uint32_t want_left = joined ? 0 : bit;
226 if ((wanted_left_ & bit) != want_left) {
227 wanted_left_ ^= bit;
228 // Mute before leaving and unmute before joining, so nothing that arrives
229 // while joined is dropped.
230 mute(line, !joined);
231 failing_ &= ~bit;
232 retry_at_ns_[line] = 0;
233 }
234 // A change still in flight has to be followed up even when the socket already is
235 // as wanted: it may be the opposite one, to be withdrawn or its answer collected.
236 if (((left_ & bit) == want_left && pending_op_[line] == NO_REQUEST) || now_ns < retry_at_ns_[line]) {
237 return;
238 }
239 const int error = change_membership(line, joined);
240 if (error == MEMBERSHIP_PENDING) {
241 return; // the receiver thread has not answered yet; ask again next poll
242 }
243 if (error != 0) {
244 retry_at_ns_[line] = now_ns + MEMBERSHIP_RETRY_NS;
245 if (!(failing_ & bit)) {
246 failing_ |= bit;
247 LOG_ERROR("feed {}: line {} could not {} its group, error {}; retrying every {} ms", name_, line,
248 joined ? "rejoin" : "leave", error, MEMBERSHIP_RETRY_NS / 1'000'000);
249 }
250 return;
251 }
252 left_ = (left_ & ~bit) | want_left;
253 failing_ &= ~bit;
254 }

260 template <typename OnPacket>
261 uint32_t drain(OnPacket &&on_packet) {
262 uint32_t handled = 0;
263 for (; handled < max_packets_per_poll_; ++handled) {
264 auto [entry, size] = packets_.read(cursor_);
265 if (!entry) {
266 break;
267 }
268 if (size < 1) {
269 continue;
270 }
271 on_packet(entry[0], entry + 1, static_cast<size_t>(size - 1));
272 }
273 return handled;
274 }

308 virtual int change_membership(uint8_t line, bool join) {
309 if (!receivers_[line] || !running_.load(std::memory_order_acquire)) {
310 return 0; // no thread running: nothing is open that could be changed
311 }
312 auto &request = request_[line];
313 uint8_t &pending = pending_op_[line]; // what this thread last handed over; NO_REQUEST for none
314 const uint8_t op = join ? REQUEST_JOIN : REQUEST_LEAVE;
315 for (;;) {
316 if (pending == NO_REQUEST) {
317 pending = op;
318 request.store(op, std::memory_order_release);
319 return MEMBERSHIP_PENDING;
320 }
321 const uint8_t state = request.load(std::memory_order_acquire);
322 if (state == REQUEST_DONE || state == REQUEST_FAILED) {
323 request.store(NO_REQUEST, std::memory_order_relaxed);
324 const uint8_t handed = std::exchange(pending, NO_REQUEST);
325 // The socket has changed as it was handed, whatever is wanted now: record it
326 // here, or a failed change after this one would be retried from a stale state.
327 if (state == REQUEST_DONE) {
328 const uint32_t bit = 1u << line;
329 left_ = handed == REQUEST_LEAVE ? (left_ | bit) : (left_ & ~bit);
330 }
331 if (handed == op) {
332 return state == REQUEST_DONE ? 0 : EIO;
333 }
334 continue; // the answer to the opposite change: now hand over the wanted one
335 }
336 if (state == pending && pending != op) {
337 uint8_t untouched = pending;
338 if (request.compare_exchange_strong(untouched, NO_REQUEST, std::memory_order_acq_rel)) {
339 // Never made, so the socket is still as it was before it: as `op` asks.
340 pending = NO_REQUEST;
341 return 0;
342 }
343 // The thread took it between the read and the withdrawal: wait for it.
344 }
345 return MEMBERSHIP_PENDING;
346 }

363 static bool pin_current_thread(int cpu, const std::string &name) {
364 if (cpu < 0) {
365 return false;
366 }
367 bool ok = false;
368#if defined(__linux__)
369 cpu_set_t set;
370 CPU_ZERO(&set);
371 if (cpu < CPU_SETSIZE) {
372 CPU_SET(cpu, &set);
373 ok = pthread_setaffinity_np(pthread_self(), sizeof(set), &set) == 0;
374 }
375#elif defined(_WIN32)
376 ok = cpu < 64 && SetThreadAffinityMask(GetCurrentThread(), DWORD_PTR{1} << cpu) != 0;
377#endif
378 if (ok) {
379 LOG_INFO("feed {}: receiver thread pinned to CPU core {}", name, cpu);
380 } else {
381 LOG_WARN("feed {}: could not pin the receiver thread to CPU core {}; it will sleep between sweeps", name,
382 cpu);
383 }
384 return ok;
385 }

389 void receive_loop() {
390 // Spin only on a core this thread owns: a failed pin leaves it sharing one.
391 const bool spin = pin_current_thread(receiver_cpu_, name_);
392 std::vector<uint8_t> buffer(MAX_DATAGRAM);
393 std::array<uint64_t, MAX_LINES> next_error_log_ns{};
394 constexpr int MAX_BURST = 64; // per line per sweep, so one hot line cannot starve the rest
395 while (running_.load(std::memory_order_acquire)) {
396 bool idle = true;
397 for (const uint8_t line : lines_) {
398 auto &receiver = *receivers_[line];
399 auto &request = request_[line];
400 // Taking the request with a compare-exchange lets change_membership()
401 // withdraw one this thread has not reached, so only one of them wins.
402 if (uint8_t op = request.load(std::memory_order_acquire);
403 (op == REQUEST_JOIN || op == REQUEST_LEAVE) &&
404 request.compare_exchange_strong(op, REQUEST_BUSY, std::memory_order_acq_rel)) {
405 bool ok = true;
406 if (op == REQUEST_JOIN) {
407 ok = receiver.open();
408 } else {
409 receiver.close();
410 }
411 request.store(ok ? REQUEST_DONE : REQUEST_FAILED, std::memory_order_release);
412 }
413 if (!receiver.is_open()) {
414 continue;
415 }
416 for (int i = 0; i < MAX_BURST; ++i) {
417 const auto n = receiver.receive(buffer);
418 if (n < 0) {
419 // A failing socket looks like a quiet feed otherwise. slick-socket
420 // counts the errors but does not say which; it keeps the socket open.
421 const uint64_t total = receive_errors_.fetch_add(1, std::memory_order_relaxed) + 1;
422 const auto now = static_cast<uint64_t>(
423 std::chrono::duration_cast<std::chrono::nanoseconds>(
424 std::chrono::steady_clock::now().time_since_epoch()).count());
425 if (now >= next_error_log_ns[line]) {
426 next_error_log_ns[line] = now + RECEIVE_ERROR_LOG_NS;
427 LOG_ERROR("feed {}: line {} failed to receive ({} read errors so far on the feed)", name_,
428 line, total);
429 }
430 break;
431 }
432 if (n == 0) {
433 break;
434 }
435 idle = false;
436 deliver(line, buffer.data(), static_cast<size_t>(n));
437 }
438 }
439 if (idle && !spin) {
440 std::this_thread::sleep_for(RECEIVER_IDLE_SLEEP);
441 }
442 }
443 }
