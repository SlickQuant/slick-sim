130 bool deliver(uint8_t line, const uint8_t *data, size_t size) {
131 assert(line < MAX_LINES);
132 if (muted_.load(std::memory_order_relaxed) & (1u << line)) {
133 return false;
134 }
135 const auto n = static_cast<uint32_t>(size + 1);
136 if (n > packets_.size()) [[unlikely]] {
137 // A queue too small for one packet is too small for all of them: log the
138 // first, count the rest, rather than flood the log at the feed's rate.
139 if (oversized_.fetch_add(1, std::memory_order_relaxed) == 0) {
140 LOG_ERROR("feed {}: line {} packet of {} bytes exceeds the {} byte queue; dropping it and any like it",
141 name_, line, size, packets_.size());
142 }
143 return false;
144 }
145 const uint64_t index = packets_.reserve(n);
146 uint8_t *entry = packets_[index];
147 entry[0] = line;
148 std::memcpy(entry + 1, data, size);
149 packets_.publish(index, n);
150 return true;
151 }

154 void mute(uint8_t line, bool muted) noexcept {
155 assert(line < MAX_LINES);
156 const uint32_t bit = 1u << line;
157 if (muted) {
158 muted_.fetch_or(bit, std::memory_order_relaxed);
159 } else {
160 muted_.fetch_and(~bit, std::memory_order_relaxed);
161 }
162 }

173 void set_joined(uint8_t line, bool joined, uint64_t now_ns) {
174 assert(line < MAX_LINES);
175 const uint32_t bit = 1u << line;
176 const uint32_t want_left = joined ? 0 : bit;
177 if ((wanted_left_ & bit) != want_left) {
178 wanted_left_ ^= bit;
179 // Mute before leaving and unmute before joining, so nothing that arrives
180 // while joined is dropped.
181 mute(line, !joined);
182 failing_ &= ~bit;
183 retry_at_ns_[line] = 0;
184 }
185 if ((left_ & bit) == want_left || now_ns < retry_at_ns_[line]) {
186 return;
187 }
188 if (const int error = change_membership(line, joined); error != 0) {
189 retry_at_ns_[line] = now_ns + MEMBERSHIP_RETRY_NS;
190 if (!(failing_ & bit)) {
191 failing_ |= bit;
192 LOG_ERROR("feed {}: line {} could not {} its group, error {}; retrying every {} ms", name_, line,
193 joined ? "rejoin" : "leave", error, MEMBERSHIP_RETRY_NS / 1'000'000);
194 }
195 return;
196 }
197 left_ ^= bit;
198 failing_ &= ~bit;
199 }

205 template <typename OnPacket>
206 uint32_t drain(OnPacket &&on_packet) {
207 uint32_t handled = 0;
208 for (; handled < max_packets_per_poll_; ++handled) {
209 auto [entry, size] = packets_.read(cursor_);
210 if (!entry) {
211 break;
212 }
213 if (size < 1) {
214 continue;
215 }
216 on_packet(entry[0], entry + 1, static_cast<size_t>(size - 1));
217 }
218 return handled;
219 }

256 int set_membership(bool join) {
257 if (!is_running() || socket_ == invalid_socket) {
258 return 0;
259 }
260 ip_mreq mreq{};
261 if (inet_pton(AF_INET, config_.multicast_address.c_str(), &mreq.imr_multiaddr) != 1) {
262 return -1; // start() could not have joined it either
263 }
264 // As slick-socket joined it: "0.0.0.0" or an unparsable interface is any.
265 if (inet_pton(AF_INET, config_.interface_address.c_str(), &mreq.imr_interface) != 1) {
266 mreq.imr_interface.s_addr = INADDR_ANY;
267 }
268 if (setsockopt(socket_, IPPROTO_IP, join ? IP_ADD_MEMBERSHIP : IP_DROP_MEMBERSHIP,
269 reinterpret_cast<const char *>(&mreq), sizeof(mreq)) != 0) {
270#if defined(_WIN32) || defined(_WIN64)
271 const int error = WSAGetLastError();
272#else
273 const int error = errno;
274#endif
275 return error != 0 ? error : -1;
276 }
277 LOG_INFO("{}: {} group {}:{}", name_, join ? "rejoined" : "left", config_.multicast_address, config_.port);
278 return 0;
