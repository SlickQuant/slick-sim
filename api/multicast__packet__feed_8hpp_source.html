121 bool deliver(uint8_t line, const uint8_t *data, size_t size) {
122 assert(line < MAX_LINES);
123 if (muted_.load(std::memory_order_relaxed) & (1u << line)) {
124 return false;
125 }
126 const auto n = static_cast<uint32_t>(size + 1);
127 if (n > packets_.size()) [[unlikely]] {
128 // A queue too small for one packet is too small for all of them: log the
129 // first, count the rest, rather than flood the log at the feed's rate.
130 if (oversized_.fetch_add(1, std::memory_order_relaxed) == 0) {
131 LOG_ERROR("feed {}: line {} packet of {} bytes exceeds the {} byte queue; dropping it and any like it",
132 name_, line, size, packets_.size());
133 }
134 return false;
135 }
136 const uint64_t index = packets_.reserve(n);
137 uint8_t *entry = packets_[index];
138 entry[0] = line;
139 std::memcpy(entry + 1, data, size);
140 packets_.publish(index, n);
141 return true;
142 }
143
145 void mute(uint8_t line, bool muted) noexcept {
146 assert(line < MAX_LINES);
147 const uint32_t bit = 1u << line;
148 if (muted) {
149 muted_.fetch_or(bit, std::memory_order_relaxed);
150 } else {
151 muted_.fetch_and(~bit, std::memory_order_relaxed);

159 template <typename OnPacket>
160 uint32_t drain(OnPacket &&on_packet) {
161 uint32_t handled = 0;
162 for (; handled < max_packets_per_poll_; ++handled) {
163 auto [entry, size] = packets_.read(cursor_);
164 if (!entry) {
165 break;
166 }
167 if (size < 1) {
168 continue;
169 }
170 on_packet(entry[0], entry + 1, static_cast<size_t>(size - 1));
171 }
172 return handled;
173 }
