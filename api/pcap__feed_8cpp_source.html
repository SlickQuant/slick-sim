38uint64_t parse_utc_time_ns(const std::string &text) {
39 int year, month, day, hour, minute, second;
40 int consumed = 0;
41 if (std::sscanf(text.c_str(), "%d-%d-%dT%d:%d:%d%n", &year, &month, &day, &hour, &minute, &second, &consumed) != 6) {
42 throw std::runtime_error(std::format("'{}' is not a UTC time like 2026-11-06T14:00:00Z", text));
43 }
44 uint64_t fraction_ns = 0;
45 size_t pos = static_cast<size_t>(consumed);
46 if (pos < text.size() && text[pos] == '.') {
47 ++pos;
48 uint64_t scale = 100'000'000;
49 while (pos < text.size() && text[pos] >= '0' && text[pos] <= '9') {
50 fraction_ns += static_cast<uint64_t>(text[pos] - '0') * scale;
51 scale /= 10;
52 ++pos;
53 }
54 }
55 if (pos < text.size() && text[pos] == 'Z') {
56 ++pos;
57 }
58 if (pos != text.size() || year < 1970 || month < 1 || month > 12 || day < 1 || day > 31 || hour < 0 ||
59 hour > 23 || minute < 0 || minute > 59 || second < 0 || second > 59) {
60 throw std::runtime_error(std::format("'{}' is not a UTC time like 2026-11-06T14:00:00Z", text));
61 }
62 std::tm tm{};
63 tm.tm_year = year - 1900;
64 tm.tm_mon = month - 1;
65 tm.tm_mday = day;
66 tm.tm_hour = hour;
67 tm.tm_min = minute;
68 tm.tm_sec = second;
69 const time_t seconds = timegm(&tm);
70 // timegm() normalizes an impossible date (Feb 31 -> Mar 3); reject it instead.
71 if (seconds == static_cast<time_t>(-1)) {
72 throw std::runtime_error(std::format("'{}' is not a time this system can represent", text));
73 }
74 if (tm.tm_mday != day || tm.tm_mon != month - 1) {
75 throw std::runtime_error(std::format("'{}' is not a valid date", text));
76 }
77 return static_cast<uint64_t>(seconds) * 1'000'000'000ull + fraction_ns;
78}

80PcapFeedConfig PcapFeedConfig::from_json(const nlohmann::json &config, const std::string &feed) {
81 PcapFeedConfig c;
82 c.root = config.value("pcap_root", std::string());
83 c.begin_ns = time_from_json(config, "begin", feed);
84 c.end_ns = time_from_json(config, "end", feed);
85 if (c.end_ns <= c.begin_ns) {
86 throw std::runtime_error(std::format("pcap feed {}: 'end' must be after 'begin'", feed));
87 }
88 c.speed = config.value("speed", 1.0);
89 if (!(c.speed >= 0) || !std::isfinite(c.speed)) {
90 throw std::runtime_error(std::format("pcap feed {}: 'speed' must be 0 (unpaced) or above", feed));
91 }
92 if (!config.contains("control_channel") || !config["control_channel"].contains("port")) {
93 throw std::runtime_error(std::format(
94 "pcap feed {}: needs control_channel.port - clients start the replay over it, and it waits for them", feed));
95 }
96 const auto port = config["control_channel"]["port"].get<int64_t>();
97 if (port < 0 || port > 65535) {
98 throw std::runtime_error(std::format("pcap feed {}: control_channel.port {} is not a port", feed, port));
99 }
100 c.control_port = static_cast<uint16_t>(port);
101 c.control_max_connections = config["control_channel"].value("max_connections", c.control_max_connections);
102 return c;
103}

105std::vector<PcapLine> parse_pcap_lines(const nlohmann::json &config,
106 std::span<const std::pair<const char *, uint8_t>> keys) {
107 std::vector<PcapLine> lines;
108 for (const auto &[key, line] : keys) {
109 if (!config.contains(key)) {
110 continue;
111 }
112 const auto &group = config[key];
113 PcapLine l;
114 l.line = line;
115 l.address = group.at("address").get<std::string>();
116 l.port = group.at("port").get<uint16_t>();
117 if (group.contains("path")) {
118 l.paths.push_back(group["path"].get<std::string>());
119 }
120 if (group.contains("paths")) {
121 for (const auto &path : group["paths"]) {
122 l.paths.push_back(path.get<std::string>());
123 }
124 }
125 lines.push_back(std::move(l));
126 }
127 return lines;
128}

169void PcapFeedBase::enter(FeedState next, uint64_t now_ns, std::string message) {
170 if (state_ == next && message == message_) {
171 return;
172 }
173 state_ = next;
174 message_ = std::move(message);
175 if (next == FeedState::ERROR) {
176 LOG_ERROR("pcap feed {}: {}", name_, message_);
177 } else {
178 LOG_INFO("pcap feed {}: {}{}{}", name_, to_string(next), message_.empty() ? "" : " - ", message_);
179 }
180 // What the handler made of the capture, where a gap or a malformed packet would show.
181 if (next == FeedState::READY || next == FeedState::FINISHED || next == FeedState::ERROR) {
182 if (const auto detail = describe(); !detail.empty()) {
183 LOG_INFO("pcap feed {}: {}", name_, detail);
184 }
185 if (const auto cut = source_.truncated_frames(); cut != 0) {
186 LOG_WARN("pcap feed {}: {} captured frames were cut short or malformed and skipped", name_, cut);
187 }

234void PcapFeedBase::feed(uint64_t now_ns) {
235 PcapReplaySource::Item item;
236
237 if (state_ == FeedState::LOADING) {
238 if (!source_.positioned()) {
239 return;
240 }
241 if (const auto error = source_.error(); !error.empty()) {
242 enter(FeedState::ERROR, now_ns, error);
243 return;
244 }
245 enter(FeedState::SYNCING, now_ns);
246 }
247
248 if (state_ == FeedState::SYNCING) {
249 // Unpaced and unstamped: the book is being built, and nothing goes out.
250 event_time_ns_ = now_ns;
251 for (uint32_t i = 0; i < MAX_PACKETS_PER_POLL && source_.peek(item); ++i) {
252 current_capture_ns_ = item.ts_ns;
253 on_packet(item.line, item.data, item.size, now_ns);
254 source_.pop();
255 if (synchronized()) {
256 on_timer(now_ns);
257 enter(FeedState::READY, now_ns);
258 break;
259 }
260 }
261 if (state_ == FeedState::SYNCING) {
262 on_timer(now_ns);
263 if (synchronized()) {
264 enter(FeedState::READY, now_ns);
265 } else {
266 if (source_.finished()) {
267 const auto error = source_.error();
268 enter(FeedState::ERROR, now_ns,
269 error.empty() ? "the capture ended before the book was synchronized; widen the window "
270 "or check that the snapshot capture covers its start"
271 : error);
272 }
273 return;
274 }
275 }
276 }
277
278 if (state_ == FeedState::READY) {
279 if (!start_requested_) {
280 return;
281 }
282 // The held book is published from now, even if no packet is left to replay.
283 gate_->open();
284 if (!source_.peek(item)) {
285 if (source_.finished()) {
286 if (const auto error = source_.error(); !error.empty()) {
287 enter(FeedState::ERROR, now_ns, error);
288 } else {
289 enter(FeedState::FINISHED, now_ns);
290 }
291 }
292 return;
293 }
294 // The capture time of the next packet is what the wall clock starts from.
295 pacer_.anchor(item.ts_ns, now_ns);
296 enter(FeedState::REPLAYING, now_ns);
297 }
298
299 if (state_ == FeedState::REPLAYING) {
300 for (uint32_t i = 0; i < MAX_PACKETS_PER_POLL && source_.peek(item); ++i) {
301 if (!pacer_.is_due(item.ts_ns, now_ns)) {
302 break;
303 }
304 current_capture_ns_ = item.ts_ns;
305 event_time_ns_ = pacer_.rebase_ns(item.ts_ns, now_ns);
306 on_packet(item.line, item.data, item.size, now_ns);
307 source_.pop();
308 }
309 // The handler's timers fire now, not at the last packet's time.
310 event_time_ns_ = now_ns;
311 on_timer(now_ns);
312 if (source_.finished()) {
313 if (const auto error = source_.error(); !error.empty()) {
314 enter(FeedState::ERROR, now_ns, error);
315 } else {
316 enter(FeedState::FINISHED, now_ns);
317 }
318 }
319 }
320}
