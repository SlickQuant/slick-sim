9bool JsonParser::can_handle(const uint8_t* data, size_t length) const {
10 if (length == 0) return false;
11
12 // Look for JSON indicators in first few bytes
13 std::string str(reinterpret_cast<const char*>(data), std::min(length, size_t(20)));
14
15 // Skip whitespace
16 auto first_non_space = str.find_first_not_of(" \t\n\r");
17 if (first_non_space == std::string::npos) return false;
18
19 return str[first_non_space] == '{' || str[first_non_space] == '[';

167bool JsonParser::is_complete_json_message(const std::string& buffer) const {
168 if (buffer.empty()) return false;
169
170 int brace_count = 0;
171 int bracket_count = 0;
172 bool in_string = false;
173 bool escape_next = false;
174
175 for (char c : buffer) {
176 if (escape_next) {
177 escape_next = false;
178 continue;
179 }
180
181 if (c == '\\' && in_string) {
182 escape_next = true;
183 continue;
184 }
185
186 if (c == '"') {
187 in_string = !in_string;
188 continue;
189 }
190
191 if (!in_string) {
192 switch (c) {
193 case '{': brace_count++; break;
194 case '}': brace_count--; break;
195 case '[': bracket_count++; break;
196 case ']': bracket_count--; break;
197 }
198 }
199 }
200
201 return brace_count == 0 && bracket_count == 0;
202}
