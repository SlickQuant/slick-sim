30template<size_t N>
31inline bool copy_wire_field(char (&dst)[N], std::string_view src) noexcept {
32 const auto len = std::min(src.size(), N - 1);
33 if (len) {
34 // Guarded because a default-constructed string_view has a null data()
35 // pointer, and memcpy from null is undefined even for zero bytes.
36 std::memcpy(dst, src.data(), len);
37 }
38 std::memset(dst + len, 0, N - len);
39 return len < src.size();
40}
