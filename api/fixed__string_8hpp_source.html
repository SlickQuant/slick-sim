23template<size_t N>
24inline bool copy_wire_field(char (&dst)[N], std::string_view src) noexcept {
25 const auto len = std::min(src.size(), N - 1);
26 if (len) {
27 // Guarded because a default-constructed string_view has a null data()
28 // pointer, and memcpy from null is undefined even for zero bytes.
29 std::memcpy(dst, src.data(), len);
30 }
31 std::memset(dst + len, 0, N - len);
32 return len < src.size();
33}
