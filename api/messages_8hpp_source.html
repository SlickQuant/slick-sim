328// setOrderIdentity() copies an order's OrderIdentity over the head of an
329// OrderResponse in one go, which is only valid while the two layouts agree byte
330// for byte. Adding, reordering or resizing a field at either end fails the build
331// here rather than silently shifting the ids on the wire.
332static_assert(std::is_standard_layout_v<OrderResponse>);
333static_assert(offsetof(OrderResponse, symbol) == offsetof(OrderIdentity, symbol));
334static_assert(offsetof(OrderResponse, user_id) == offsetof(OrderIdentity, user_id));
335static_assert(offsetof(OrderResponse, client_order_id) == offsetof(OrderIdentity, client_order_id));
336static_assert(offsetof(OrderResponse, order_id) == offsetof(OrderIdentity, order_id));
337static_assert(sizeof(OrderResponse::symbol) == decltype(OrderIdentity::symbol)::buffer_size());
338static_assert(sizeof(OrderResponse::user_id) == decltype(OrderIdentity::user_id)::buffer_size());
339static_assert(sizeof(OrderResponse::client_order_id) == decltype(OrderIdentity::client_order_id)::buffer_size());
340static_assert(sizeof(OrderResponse::order_id) == decltype(OrderIdentity::order_id)::buffer_size());
341static_assert(offsetof(OrderResponse, error_message) == sizeof(OrderIdentity));
342
345inline void setOrderIdentity(OrderResponse &response, const Order &order) noexcept {
346 std::memcpy(&response, static_cast<const OrderIdentity *>(&order), sizeof(OrderIdentity));
347}
