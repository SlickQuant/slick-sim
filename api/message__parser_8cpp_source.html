70std::unique_ptr<MessageParser> MessageParserFactory::create_parser_from_data(const uint8_t* data, size_t length) {
71 if (length == 0) return nullptr;
72
73 // Try each available parser to see which one can handle the data
74 std::vector<std::unique_ptr<MessageParser>> parsers;
75 parsers.push_back(std::make_unique<JsonParser>());
76 parsers.push_back(std::make_unique<FixParser>());
77 parsers.push_back(std::make_unique<SbeParser>());
78
79 for (auto& parser : parsers) {
80 if (parser->can_handle(data, length)) {
