24// One side's contribution to "l" (changed/added, in current order) and "r"
25// (indices into `prev` that are no longer present in `curr`).
26//
27// Both lists carry the live venue's own ordering, which a client is entitled to
28// rely on: "l" in book order (best price first), and "r" in ascending *price*
29// order - which is descending index on the bid side, whose array runs from the
30// highest price down. Hence `descending_index`, set for bids only.
31//
32// Emitting bid removals in ascending index order instead broke any client that
33// splices them off one at a time, because each deletion shifts the entries
34// behind it and the next index then resolves one slot too far. Hyperliquid's
35// own bid removals never arrive in that order, so such a client is correct
36// against the venue and desynced only here - the failure surfaced as a stale
37// level resting above the opposite side.
38void diff_side(const slick::sim::md_publisher::L2Levels& prev,
39 const slick::sim::md_publisher::L2Levels& curr,
40 bool descending_index,
41 nlohmann::json& l_out, nlohmann::json& r_out) {
42 std::unordered_map<price_t, qty_t> prev_by_price;
43 prev_by_price.reserve(prev.size());
44 for (const auto& [price, qty] : prev) {
45 prev_by_price.emplace(price, qty);
46 }
47
48 std::unordered_set<price_t> curr_prices;
49 curr_prices.reserve(curr.size());
50 for (const auto& [price, qty] : curr) {
51 curr_prices.insert(price);
52 }
53
54 l_out = nlohmann::json::array();
55 for (const auto& [price, qty] : curr) {
56 auto it = prev_by_price.find(price);
57 if (it == prev_by_price.end() || it->second != qty) {
58 l_out.push_back({{"p", format_price(price)}, {"s", format_qty(qty)}});
59 }
60 }
61
62 r_out = nlohmann::json::array();
63 if (descending_index) {
64 for (size_t i = prev.size(); i-- > 0; ) {
65 if (!curr_prices.contains(prev[i].first)) {
66 r_out.push_back(i);
67 }
68 }
69 } else {
70 for (size_t i = 0; i < prev.size(); ++i) {
71 if (!curr_prices.contains(prev[i].first)) {
72 r_out.push_back(i);
73 }
74 }
75 }
76}

78std::string base64_encode(const std::vector<uint8_t>& data) {
79 if (data.empty()) return {};

92void split_book_snapshot(const MDLevel* levels, uint32_t num_bid, uint32_t num_ask,

96 const uint32_t ask_count = std::min(num_ask, L2_PUBLISH_DEPTH);
97
98 bids.clear();
99 asks.clear();
100 curr_bid.clear();
101 curr_ask.clear();
102 bids.reserve(bid_count);
103 asks.reserve(ask_count);
104 curr_bid.reserve(bid_count);
105 curr_ask.reserve(ask_count);
106
107 auto emit = [](const MDLevel& level, nlohmann::json::array_t& out, L2Levels& baseline) {
108 out.push_back({
109 {"px", format_price(level.price)},
110 {"sz", format_qty(level.qty)},
111 {"n", level.num_orders}
112 });
113 baseline.emplace_back(level.price, level.qty);
114 };
115
116 for (uint32_t i = 0; i < bid_count; ++i) {
117 emit(levels[i], bids, curr_bid);
118 }
119 // The ask block starts after every bid the frame carries, not after the ones
120 // published: truncating the bids must not walk the read cursor back.
121 for (uint32_t i = 0; i < ask_count; ++i) {
122 emit(levels[num_bid + i], asks, curr_ask);
123 }
124}

138std::string encode_l2_diff(const nlohmann::json& payload) {
139 const std::string plain = payload.dump();
140
141 z_stream stream{};
142 if (deflateInit2(&stream, Z_DEFAULT_COMPRESSION, Z_DEFLATED, -MAX_WBITS, 8, Z_DEFAULT_STRATEGY) != Z_OK) {
143 throw std::runtime_error("Failed to initialize raw deflate encoder");
144 }
145 stream.next_in = reinterpret_cast<Bytef*>(const_cast<char*>(plain.data()));
146 stream.avail_in = static_cast<uInt>(plain.size());
147
148 std::vector<uint8_t> compressed(deflateBound(&stream, static_cast<uLong>(plain.size())));
149 stream.next_out = compressed.data();
150 stream.avail_out = static_cast<uInt>(compressed.size());
151
152 int result = deflate(&stream, Z_FINISH);
153 size_t produced = compressed.size() - stream.avail_out;
154 deflateEnd(&stream);
155 if (result != Z_STREAM_END) {
156 throw std::runtime_error("Failed to deflate l2 diff payload");
157 }
158 compressed.resize(produced);
159
160 return base64_encode(compressed);
161}
