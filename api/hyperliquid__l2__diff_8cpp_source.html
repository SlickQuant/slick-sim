22// One side's contribution to "l" (changed/added, in current order) and "r"
23// (indices into `prev` that are no longer present in `curr`).
24void diff_side(const slick::sim::md_publisher::L2Levels& prev,
25 const slick::sim::md_publisher::L2Levels& curr,
26 nlohmann::json& l_out, nlohmann::json& r_out) {
27 std::unordered_map<price_t, qty_t> prev_by_price;
28 prev_by_price.reserve(prev.size());
29 for (const auto& [price, qty] : prev) {
30 prev_by_price.emplace(price, qty);
31 }
32
33 std::unordered_set<price_t> curr_prices;
34 curr_prices.reserve(curr.size());
35 for (const auto& [price, qty] : curr) {
36 curr_prices.insert(price);
37 }
38
39 l_out = nlohmann::json::array();
40 for (const auto& [price, qty] : curr) {
41 auto it = prev_by_price.find(price);
42 if (it == prev_by_price.end() || it->second != qty) {
43 l_out.push_back({{"p", format_price(price)}, {"s", format_qty(qty)}});
44 }
45 }
46
47 r_out = nlohmann::json::array();
48 for (size_t i = 0; i < prev.size(); ++i) {
49 if (!curr_prices.contains(prev[i].first)) {
50 r_out.push_back(i);
51 }
52 }
53}

55std::string base64_encode(const std::vector<uint8_t>& data) {
56 if (data.empty()) return {};

81std::string encode_l2_diff(const nlohmann::json& payload) {
82 const std::string plain = payload.dump();
83
84 z_stream stream{};
85 if (deflateInit2(&stream, Z_DEFAULT_COMPRESSION, Z_DEFLATED, -MAX_WBITS, 8, Z_DEFAULT_STRATEGY) != Z_OK) {
86 throw std::runtime_error("Failed to initialize raw deflate encoder");
87 }
88 stream.next_in = reinterpret_cast<Bytef*>(const_cast<char*>(plain.data()));
89 stream.avail_in = static_cast<uInt>(plain.size());
90
91 std::vector<uint8_t> compressed(deflateBound(&stream, static_cast<uLong>(plain.size())));
92 stream.next_out = compressed.data();
93 stream.avail_out = static_cast<uInt>(compressed.size());
94
95 int result = deflate(&stream, Z_FINISH);
96 size_t produced = compressed.size() - stream.avail_out;
97 deflateEnd(&stream);
98 if (result != Z_STREAM_END) {
99 throw std::runtime_error("Failed to deflate l2 diff payload");
100 }
101 compressed.resize(produced);
102
103 return base64_encode(compressed);
104}
