164void HyperliquidPublisher::handle_message(wsT* ws, std::string_view message) {
165 try {
166 json msg = json::parse(message);
167
168 std::string method = msg.value("method", "");
169 if (method == "ping") {
170 json pong = {{"channel", "pong"}};
171 ws->send(pong.dump(), uWS::OpCode::TEXT);
172 return;
173 }
174
175 if (method != "subscribe" && method != "unsubscribe") {
176 send_error(ws, std::format("method `{}` is not supported", method));
177 return;
178 }
179
180 if (!msg.contains("subscription")) {
181 send_error(ws, "missing `subscription` field");
182 return;
183 }
184
185 const auto& sub = msg["subscription"];
186 std::string type = sub.value("type", "");
187
188 if (type == "l2Book") {
189 std::string coin = sub.value("coin", "");
190 if (coin.empty()) { send_error(ws, "missing `coin` in subscription"); return; }
191 if (method == "subscribe") {
192 subscribe_channel(ws, HyperliquidChannel::L2_BOOK, coin);
193 }
194 } else if (type == "trades") {
195 std::string coin = sub.value("coin", "");
196 if (coin.empty()) { send_error(ws, "missing `coin` in subscription"); return; }
197 if (method == "subscribe") {
198 subscribe_channel(ws, HyperliquidChannel::TRADES, coin);
199 }
200 } else if (type == "l2") {
201 // Undocumented channel — coin is field "c", not "coin".
202 std::string coin = sub.value("c", "");
203 if (coin.empty()) { send_error(ws, "missing `c` in subscription"); return; }
204 if (method == "subscribe") {
205 // Real Hyperliquid acks "l2" subscriptions immediately by
206 // echoing the subscription back, ahead of any snapshot data.
207 json ack = {
208 {"channel", "subscriptionResponse"},
209 {"data", {{"method", "subscribe"}, {"subscription", sub}}}
210 };
211 ws->send(ack.dump(), uWS::OpCode::TEXT);
212 subscribe_channel(ws, HyperliquidChannel::L2, coin);
213 }
214 } else if (type == "heartbeat") {
215 ws->getUserData()->subscribed_heartbeat = true;
216 json rsp = {{"channel", "subscriptions"}, {"data", {{"type", "heartbeat"}}}};
217 ws->send(rsp.dump(), uWS::OpCode::TEXT);
218 } else {
219 send_error(ws, std::format("subscription type `{}` is not supported", type));
220 }
221 } catch (const json::parse_error& e) {
222 send_error(ws, std::string("invalid JSON: ") + e.what());
223 }
224}

249void HyperliquidPublisher::publish_subscription_response(MarketDataUpdate* update) {
250 auto* response = reinterpret_cast<MDSubscriptionResponse*>(update->data);
251 uint8_t ch = response->channel;
252 auto& channel_sub = subscription_info_[ch];
253 auto it = channel_sub.find(update->symbol);
254 if (it == channel_sub.end() || it->second.empty()) return;
255
256 auto channel = static_cast<HyperliquidChannel>(ch);
257 json snapshot_msg;
258
259 if (channel == HyperliquidChannel::L2_BOOK || channel == HyperliquidChannel::L2) {
260 auto* snapshot = reinterpret_cast<BookSnapshot*>(response->data);
261 auto* levels = reinterpret_cast<MDLevel*>(snapshot->levels);
262
263 json::array_t bids, asks;
264 L2Levels curr_bid, curr_ask;
265 curr_bid.reserve(snapshot->num_bid);
266 curr_ask.reserve(snapshot->num_ask);
267 for (uint32_t i = 0; i < snapshot->num_bid; ++i) {
268 bids.push_back({
269 {"px", std::to_string(to_price_double(levels[i].price))},
270 {"sz", std::to_string(to_qty_double(levels[i].qty))},
271 {"n", levels[i].num_orders}
272 });
273 curr_bid.emplace_back(levels[i].price, levels[i].qty);
274 }
275 for (uint32_t i = snapshot->num_bid; i < snapshot->num_bid + snapshot->num_ask; ++i) {
276 asks.push_back({
277 {"px", std::to_string(to_price_double(levels[i].price))},
278 {"sz", std::to_string(to_qty_double(levels[i].qty))},
279 {"n", levels[i].num_orders}
280 });
281 curr_ask.emplace_back(levels[i].price, levels[i].qty);
282 }
283
284 if (channel == HyperliquidChannel::L2_BOOK) {
285 snapshot_msg = {
286 {"channel", "l2Book"},
287 {"data", {
288 {"coin", update->symbol},
289 {"time", get_current_time_ns() / ONE_MILLISECOND_NS},
290 {"levels", {bids, asks}}
291 }}
292 };
293 } else {
294 // Undocumented "l2" channel: first delivery after subscribing is
295 // an uncompressed full snapshot under key "s" — there is nothing
296 // to diff against yet. Seed the diff baseline from it so the
297 // next routine book update can be sent as a compressed diff.
298 l2_diff_baseline_[update->symbol] = {std::move(curr_bid), std::move(curr_ask)};
299 snapshot_msg = {
300 {"channel", "l2"},
301 {"data", {{"s", {
302 {"coin", update->symbol},
303 {"time", get_current_time_ns() / ONE_MILLISECOND_NS},
304 {"levels", {bids, asks}}
305 }}}}
306 };
307 }
308 } else if (channel == HyperliquidChannel::TRADES) {
309 snapshot_msg = {
310 {"channel", "trades"},
311 {"data", json::array()}
312 };
313 } else {
314 return;
315 }
316
317 for (auto* ws : it->second) {
318 auto* ud = ws->getUserData();
319 auto& sub = ud->subs[ch];
320 auto it_sym = sub.pending.find(update->symbol);
321 if (it_sym != sub.pending.end()) {
322 sub.pending.erase(it_sym);
323 sub.received.emplace_back(update->symbol);
324 ws->send(snapshot_msg.dump(), uWS::OpCode::TEXT);
325 LOG_DEBUG("HyperliquidPublisher sent subscription snapshot for {}", update->symbol);
326
327 if (sub.pending.empty()) {
328 json confirm = {
329 {"channel", "subscriptions"},
330 {"data", {{"subscriptions", sub.received}}}
331 };
332 ws->send(confirm.dump(), uWS::OpCode::TEXT);
333 }
334 }
335 }

338void HyperliquidPublisher::publish_book_snapshot(MarketDataUpdate* update) {
339 auto l2book_ch = static_cast<uint8_t>(HyperliquidChannel::L2_BOOK);
340 auto l2_ch = static_cast<uint8_t>(HyperliquidChannel::L2);
341 auto it_l2book = subscription_info_[l2book_ch].find(update->symbol);
342 auto it_l2 = subscription_info_[l2_ch].find(update->symbol);
343 bool has_l2book = it_l2book != subscription_info_[l2book_ch].end() && !it_l2book->second.empty();
344 bool has_l2 = it_l2 != subscription_info_[l2_ch].end() && !it_l2->second.empty();
345 if (!has_l2book && !has_l2) return;
346
347 auto* snapshot = reinterpret_cast<BookSnapshot*>(update->data);
348 auto* levels = reinterpret_cast<MDLevel*>(snapshot->levels);
349
350 json::array_t bids, asks;
351 L2Levels curr_bid, curr_ask;
352 curr_bid.reserve(snapshot->num_bid);
353 curr_ask.reserve(snapshot->num_ask);
354 for (uint32_t i = 0; i < snapshot->num_bid; ++i) {
355 bids.push_back({
356 {"px", std::to_string(to_price_double(levels[i].price))},
357 {"sz", std::to_string(to_qty_double(levels[i].qty))},
358 {"n", levels[i].num_orders}
359 });
360 curr_bid.emplace_back(levels[i].price, levels[i].qty);
361 }
362 for (uint32_t i = snapshot->num_bid; i < snapshot->num_bid + snapshot->num_ask; ++i) {
363 asks.push_back({
364 {"px", std::to_string(to_price_double(levels[i].price))},
365 {"sz", std::to_string(to_qty_double(levels[i].qty))},
366 {"n", levels[i].num_orders}
367 });
368 curr_ask.emplace_back(levels[i].price, levels[i].qty);
369 }
370 uint64_t time_ms = get_current_time_ns() / ONE_MILLISECOND_NS;
371
372 if (has_l2book) {
373 json msg = {
374 {"channel", "l2Book"},
375 {"data", {
376 {"coin", update->symbol},
377 {"time", time_ms},
378 {"levels", {bids, asks}}
379 }}
380 };
381 std::string payload = msg.dump();
382 for (auto* ws : it_l2book->second) {
383 ws->send(payload, uWS::OpCode::TEXT);
384 }
385 }
386
387 if (has_l2) {
388 auto& baseline = l2_diff_baseline_[update->symbol];
389 L2Diff diff = compute_l2_diff(baseline.first, baseline.second, curr_bid, curr_ask);
390 baseline.first = curr_bid;
391 baseline.second = curr_ask;
392
393 json diff_payload = {
394 {"c", update->symbol},
395 {"l", diff.l},
396 {"r", diff.r},
397 {"t", time_ms}
398 };
399 json msg = {
400 {"channel", "l2"},
401 {"data", {{"c", encode_l2_diff(diff_payload)}}}
402 };
403 std::string payload = msg.dump();
404 for (auto* ws : it_l2->second) {
405 ws->send(payload, uWS::OpCode::TEXT);
406 }
407 }

436void HyperliquidPublisher::check_heartbeats() {
437 auto now = std::chrono::steady_clock::now();
438 for (auto it = clients_.begin(); it != clients_.end();) {
439 auto* ws = *it;
440 auto* ud = ws->getUserData();
441 auto elapsed = std::chrono::duration_cast<std::chrono::seconds>(now - ud->last_pong);
442 if (elapsed > pong_timeout_) {
443 LOG_WARN("HyperliquidPublisher: client {:p} timed out, disconnecting", (void*)ws);
444 it = clients_.erase(it);
445 ws->close();
446 } else {
447 if (ud->subscribed_heartbeat) {
448 json hb = {
449 {"channel", "heartbeats"},
450 {"data", {{"time", format_timestamp_iso8601()}, {"counter", ud->heartbeat_count++}}}

452 ws->send(hb.dump(), uWS::OpCode::TEXT);
453 }
454 ws->send("", uWS::OpCode::PING);
455 ++it;
456 }

466void HyperliquidPublisher::unsubscribe_md(wsT* ws) {
467 auto* ud = ws->getUserData();
468 for (uint8_t ch = 0; ch < static_cast<uint8_t>(HyperliquidChannel::__COUNT__); ++ch) {
469 for (const auto& coin : ud->subs[ch].active) {
470 auto index = request_queue_.reserve();
471 auto* req = request_queue_[index];
472 memcpy(req->symbol, coin.data(), sizeof(req->symbol));
473 req->msg_type = MessageType::MD_UNSUBSCRIPTION;
474 req->md_subscription.channel = ch;
475 req->md_subscription.client = (void*)ws;
476 request_queue_.publish(index);
477
478 auto it_sym = symbol_channel_client_.find(coin);
479 if (it_sym != symbol_channel_client_.end()) {
480 auto range = it_sym->second.equal_range(ch);
481 for (auto itr = range.first; itr != range.second; ) {
482 if (itr->second == ws) { itr = it_sym->second.erase(itr); break; }
483 else ++itr;
484 }
485 if (it_sym->second.empty()) symbol_channel_client_.erase(it_sym);
486 }
487 auto& coin_subs = subscription_info_[ch][coin];
488 coin_subs.erase(ws);
489 if (ch == static_cast<uint8_t>(HyperliquidChannel::L2) && coin_subs.empty()) {
490 l2_diff_baseline_.erase(coin);
491 }
492 }
493 ud->subs[ch].active.clear();
494 ud->subs[ch].pending.clear();
495 ud->subs[ch].received.clear();
496 }
