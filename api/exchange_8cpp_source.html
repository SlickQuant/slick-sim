104void Exchange::handleNewOrderRequest(const Request &request) {
105 auto *symbol = symbol_mgr.getSymbol(request.symbol);
106 if (!symbol)
107 {
108 rejectNewOrderRequest(request, OrdRejectReason::UNKNOWN_CONTRACT);
109 }
110 else
111 {
112 auto &msg = request.add_order;
113 if (msg.client_order_id[0] != '\0') {
114 auto *order = symbol->order_book_->findOrderByClientOrderId(msg.client_order_id);
115 if (order) [[unlikely]] {
116 LOG_WARN(std::format("ClientOrderId {} already exist", msg.client_order_id));
117 rejectNewOrderRequest(request, OrdRejectReason::CLIENT_ORDER_ID_ALREADY_EXISTS);
118 return;
119 }
120 }
121 Order *order = symbol->order_book_->allocateOrder();
122 order->symbol = symbol->symbol_;
123 order->client_order_id = msg.client_order_id;
124 order->user_id = msg.user_id;
125 order->side = msg.side;
126 order->type = msg.type;
127 order->time_in_force = msg.time_in_force;
128 order->price = msg.price;
129 order->quantity = msg.qty;
130 order->leaves_quantity = msg.qty;
131 order->type = msg.type;
132 order->status = OrderStatus::PENDING_NEW;
133
134 sendOrderNewPending(order, request.time_stamp);
135
136 auto [reject_reason, trade_summaries] = symbol->addOrder(order, request.time_stamp);
137
138 if (!trade_summaries.empty()) {
139 for (const auto &summary : trade_summaries) {
140 publishTradeSummary(symbol->symbol_.c_str(), summary);
141 }
142 }
143
144 if (reject_reason != OrdRejectReason::NONE && reject_reason != OrdRejectReason::FOK_CANNOT_FILL) {
145 rejectNewOrderRequest(request, reject_reason);
146 }
147
148 if (!symbol->md_level_update_cache_.empty()) {
149 publishLevelUpdate(symbol->symbol_.c_str(), symbol->md_level_update_cache_);
150 symbol->md_level_update_cache_.clear();
151 }
152
153 if (!symbol->md_order_update_cache_.empty()) {
154 // TODO: publish MD order update
155 symbol->md_order_update_cache_.clear();
156 }
157 }

160void Exchange::handleModifyOrderRequest(const Request &request) {
161 auto *symbol = symbol_mgr.getSymbol(request.symbol);
162 if (!symbol)
163 {
164 rejectModifyOrderRequest(request, OrdRejectReason::UNKNOWN_CONTRACT);
165 }
166 else
167 {
168 Order *order;
169 if (request.modify_order.client_order_id[0] != '\0') {
170 order = symbol->findOrderByClientOrderId(request.modify_order.client_order_id);
171 }
172 else {
173 order = symbol->findOrderByOrderId(request.modify_order.order_id);
174 }
175
176 if (!order) {
177 LOG_WARN(std::format("{} OrderId {} not found for modification", symbol->symbol_, request.modify_order.order_id));
178 rejectModifyOrderRequest(request, OrdRejectReason::UNKNOWN_ORDER);
179 return;
180 }
181 order->status = OrderStatus::PENDING_REPLACE;
182 order->last_update_time = utils::get_current_time_ns();
183 sendOrderReplacePending(order, request.time_stamp);
184 auto [reject_reason, trade_summaries] = symbol->modifyOrder(order, request.modify_order.new_price, request.modify_order.new_qty, request.time_stamp);
185
186 if (!trade_summaries.empty()) {
187 for (const auto &summary : trade_summaries) {
188 publishTradeSummary(symbol->symbol_.c_str(), summary);
189 }
190 }
191
192 if (reject_reason != OrdRejectReason::NONE) {
193 rejectModifyOrderRequest(request, reject_reason);
194 }
195
196 if (!symbol->md_level_update_cache_.empty()) {
197 publishLevelUpdate(symbol->symbol_.c_str(), symbol->md_level_update_cache_);
198 symbol->md_level_update_cache_.clear();
