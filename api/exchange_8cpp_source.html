98void Exchange::processRequest()
99{
100 // process order rquests
101 auto [request, size] = request_queue_.read(order_request_cursor_);
102 if (request)
103 {
104 switch (request->msg_type)
105 {
106 case MessageType::NEW_ORDER_SINGLE:
107 {
108 handleNewOrderRequest(*request);
109 break;
110 }
111 case MessageType::ORDER_REPLACE_REQUEST:
112 {
113 handleModifyOrderRequest(*request);
114 break;
115 }
116 case MessageType::ORDER_CANCEL_REQUEST:
117 {
118 handleCancelOrderRequest(*request);
119 break;
120 }
121 case MessageType::MD_SUBSCRIPTION:
122 {
123 handleMdSubscription(*request);
124 break;
125 }
126 case MessageType::MD_UNSUBSCRIPTION:
127 {
128 handleMdUnsubscription(*request);
129 break;
130 }
131 default:
132 break;
133 }

137int Exchange::clientIdFor(const char *user_id)
138{
139 auto len = strnlen(user_id, sizeof(AddOrderMessage::user_id));
140 if (len == 0)
141 {
142 // No identity to compare, so this order can never self-match.
143 return -1;
144 }
145
146 // Heterogeneous lookup first: this runs once per new order, and the same
147 // handful of users submit all of them, so the hit path must not build a key.
148 std::string_view key(user_id, len);
149 if (auto it = client_ids_.find(key); it != client_ids_.end())
150 {
151 return it->second;
152 }
153
154 // First order from this user - only now is a key worth materialising, and it
155 // is stored inline rather than on the heap.
156 return client_ids_.emplace(key, next_client_id_++).first->second;
157}

159void Exchange::handleNewOrderRequest(const Request &request)
160{
161 auto *symbol = symbol_mgr.getSymbol(request.symbol, venue_);
162 if (!symbol)
163 {
164 rejectNewOrderRequest(request, OrdRejectReason::UNKNOWN_CONTRACT);
165 }
166 else
167 {
168 auto &msg = request.add_order;
169 if (msg.client_order_id[0] != '\0')
170 {
171 auto *order = symbol->order_book_->findOrderByClientOrderId(msg.client_order_id);
172 if (order) [[unlikely]]
173 {
174 LOG_WARN(std::format("ClientOrderId {} already exist", msg.client_order_id));
175 rejectNewOrderRequest(request, OrdRejectReason::CLIENT_ORDER_ID_ALREADY_EXISTS);
176 return;
177 }
178 }
179 Order *order = symbol->order_book_->allocateOrder();
180 order->symbol = symbol->symbol_;
181 order->client_order_id = msg.client_order_id;
182 order->user_id = msg.user_id;
183 // Identity for self-match prevention. Scoped to the authenticated user,
184 // matching how real venues scope SMP - to the account, not the connection.
185 order->client_id = clientIdFor(msg.user_id);
186 order->side = msg.side;
187 order->type = msg.type;
188 order->time_in_force = msg.time_in_force;
189 order->price = msg.price;
190 order->resetFillAccounting(msg.qty);
191 order->type = msg.type;
192 order->status = OrderStatus::PENDING_NEW;
193
194 sendOrderNewPending(order, request.time_stamp);
195
196 auto [reject_reason, trade_summaries] = symbol->addOrder(order, request.time_stamp);
197
198 if (!trade_summaries.empty())
199 {
200 for (const auto &summary : trade_summaries)
201 {
202 publishTradeSummary(symbol, summary);
203 }
204 }
205
206 if (reject_reason != OrdRejectReason::NONE && reject_reason != OrdRejectReason::FOK_CANNOT_FILL)
207 {
208 rejectNewOrderRequest(request, reject_reason);
209 }
210
211 if (!symbol->md_level_update_cache_.empty())
212 {
213 publishLevelUpdate(symbol->symbol_, symbol->md_level_update_cache_);
214 symbol->md_level_update_cache_.clear();
215 }
216
217 if (!symbol->md_order_update_cache_.empty())
218 {
219 // TODO: publish MD order update
220 symbol->md_order_update_cache_.clear();
221 }
222 }

440{
441 // The engine's own trade id comes from a single static counter shared by
442 // every engine and every venue, so it cannot serve as the public id. Each
443 // venue stamps its own sequence here, once, so a live print and the same
444 // trade replayed in a snapshot carry identical ids for every client.
445 auto trade_id = nextVenueTradeId();
446
447 auto sz = static_cast<uint32_t>(sizeof(MarketDataUpdate) + sizeof(TradeSummary) + trade_summary.num_orders * sizeof(Trade));
448 auto index = md_queue_.reserve(sz);
449 auto update = reinterpret_cast<MarketDataUpdate *>(md_queue_[index]);
450 update->type = MDUpdateType::TRADE_SUMMARY;
451 update->venue = venue_;
452 std::memcpy(update->symbol, symbol->symbol_.data(), sizeof(update->symbol));
453 auto summary = reinterpret_cast<TradeSummary *>(update->data);
454 summary->timestamp = trade_summary.timestamp;
455 summary->trade_id = trade_id;
456 summary->security_id = symbol->id_;
457 summary->aggressor_side = trade_summary.aggressor_side;
458 summary->price = trade_summary.price;
459 summary->qty = trade_summary.qty;
460 summary->num_orders = trade_summary.num_orders;
461 for (size_t i = 0; i < trade_summary.Trades.size(); ++i)

463 summary->trades[i].order_id = trade_summary.Trades[i].order_id;
464 summary->trades[i].qty = trade_summary.Trades[i].qty;
465 }
466 md_queue_.publish(index, sz);

468 symbol->recordTrade(MDTrade{
469 .trade_id = trade_id,
470 .event_time = static_cast<uint64_t>(trade_summary.timestamp),
471 .seq_num = 0,
472 .price = trade_summary.price,
473 .qty = trade_summary.qty,
474 .flags = UpdateFlags::F_NONE,
475 .side = trade_summary.aggressor_side});

478void Exchange::publishTradeSubscriptionResponse(Symbol *symbol, uint8_t channel)
479{
480 auto &history = symbol->trade_history_;
481 auto num_trades = static_cast<uint32_t>(history.size());
482
483 auto sz = static_cast<uint32_t>(sizeof(MarketDataUpdate) + sizeof(MDSubscriptionResponse) + sizeof(MDTradeSnapshotResponse) + num_trades * sizeof(MDTrade));
484 auto index = md_queue_.reserve(sz);
485 auto *update = reinterpret_cast<MarketDataUpdate *>(md_queue_[index]);
486 std::memcpy(update->symbol, symbol->symbol_.data(), sizeof(update->symbol));
487 update->venue = venue_;
488 update->type = MDUpdateType::SUB_RESPONSE;
489
490 auto *response = reinterpret_cast<MDSubscriptionResponse *>(update->data);
491 response->channel = channel;
492 response->reject_reason = MDSubscriptionRejectReason::NONE;
493
494 // A trade the book absorbed is settled history for this subscriber whatever
495 // its timestamp, because the L2 snapshot they receive is generated from that
496 // same book. Only a print the simulator never applied depends on timing, and
497 // then on whether the book's own data reaches it: `lastUpdateTime()` tracks
498 // level updates only, so it cannot stand in for trade application here.
499 auto book_time = symbol->order_book_ ? symbol->order_book_->lastUpdateTime() : 0;
500
501 auto reflected_in_book = [book_time](const MDTrade &trade) {
502 return !(trade.flags & UpdateFlags::F_NOT_IN_BOOK) || trade.event_time <= book_time;
503 };
504
505 auto *snapshot = reinterpret_cast<MDTradeSnapshotResponse *>(response->data);
506 auto *trades = snapshot->trades;
507 uint32_t num_snapshot = 0;
508 uint32_t num_update = 0;
509
510 // The consumer reads the two blocks as contiguous ranges, so they must be
511 // written that way — hence two passes rather than one. Classification is not
512 // monotonic over the history: an applied trade can follow an unapplied print
513 // that is newer than book_time, so a single pass in history order would
514 // interleave the blocks and silently swap their contents. Each pass walks the
515 // history in order, so both blocks stay chronological within themselves.
516 for (uint32_t i = 0; i < num_trades; ++i)
517 {
518 const auto *trade = history[i];
519 if (trade == nullptr) [[unlikely]]
520 {
521 continue;
522 }
523 if (reflected_in_book(*trade))
524 {
525 *trades++ = *trade;
526 ++num_snapshot;
527 }
528 }
529 for (uint32_t i = 0; i < num_trades; ++i)
530 {
531 const auto *trade = history[i];
532 if (trade == nullptr) [[unlikely]]
533 {
534 continue;
535 }
536 if (!reflected_in_book(*trade))
537 {
538 *trades++ = *trade;
539 ++num_update;
540 }
541 }
542 snapshot->num_snapshot = num_snapshot;
543 snapshot->num_update = num_update;
544 md_queue_.publish(index, sz);
545}

547void Exchange::reconcilePhantomQty(Symbol *symbol, Side side, price_t price, qty_t target_qty,
548 time_t event_time, uint64_t seq_num, bool end_event)
549{
550 auto &book = *symbol->order_book_;
551 auto [level, index] = book.getLevel(to_book_side(side), price);
552 if (!level)
553 {
554 return;
555 }
556
557 // A trade print and the venue's level update for it describe the same event.
558 // The usual case needs no help: the update carries the post-trade quantity
559 // and the book has already absorbed the trade, so the two agree and nothing
560 // moves. The exception is a trade this update is too old to have seen - the
561 // update still reports the pre-trade quantity there, and taking it at face
562 // value would add that quantity back only for the next update to remove it
563 // again. Offset by exactly the trades it has not accounted for. Timestamps
564 // that tie count as reflecting the trade: a venue emits the print and its
565 // level update together, and the sequencer applies the trade first.
566 auto unreflected = symbol->takeTradedQty(side, price, event_time);
567 if (unreflected > 0)
568 {
569 target_qty = target_qty > unreflected ? target_qty - unreflected : 0;
570 }
571
572 auto md_level_qty = book.getMDLevelQty(price);
573 if (target_qty > md_level_qty)
574 {
575 // Level qty increased, assume new orders added at the back. Create new phantom order to represent then increment
576 auto size = target_qty - md_level_qty;
577 book.addBookOrder(utils::nextOrderId(), to_book_side(side), price, size, event_time, seq_num);
578 }
579 else if (target_qty < md_level_qty)
580 {
581 // Level qty decreased. Always assume the worst case, that the quantity decreased at the back
582 auto diff = md_level_qty - target_qty;
583 for (auto it = level->orders.rbegin(); it != level->orders.rend(); ++it)
584 {
585 auto &order = *it;
586 if (symbol->findOrder(order.order_id))
587 {
588 // Belongs to the simulator's own user - never touch it.
589 continue;
590 }
591
592 auto to_reduce = std::min<qty_t>(diff, order.quantity);
593 book.modifyOrder(order.order_id, price, order.quantity - to_reduce, event_time, order.priority, seq_num, end_event && (diff == to_reduce));
594 diff -= to_reduce;
595 if (diff == 0 || book.getMDLevelQty(price) == 0)
596 {
597 break;
598 }
599 }
600 }

625void Exchange::sendOrderReplacePending(const Order *order, time_t request_time)
626{
627 auto index = response_queue_.reserve();
628 auto &response = *response_queue_[index];
629 setOrderIdentity(response, *order);
630 response.response_type = MessageType::EXECUTION_REPORT;
631 response.exec_type = ExecType::PENDING_REPLACE;
