57inline std::string to_fixed_string(int_fast64_t value) {
58 const bool negative = value < 0;
59 // Negated through the unsigned domain so the most negative value cannot overflow.
60 const uint64_t magnitude = negative
61 ? (0ULL - static_cast<uint64_t>(value))
62 : static_cast<uint64_t>(value);
63
64 const uint64_t scale = static_cast<uint64_t>(DOUBLE_MULTIPLIER);
65 uint64_t fraction = magnitude % scale;
66
67 std::string out;
68 if (negative) {
69 out.push_back('-');
70 }
71 out += std::to_string(magnitude / scale);
72
73 if (fraction == 0) {
74 return out;
75 }
76
77 char digits[8];
78 for (int i = 7; i >= 0; --i) {
79 digits[i] = static_cast<char>('0' + fraction % 10);
80 fraction /= 10;
81 }
82 std::string_view significant(digits, 8);
83 while (!significant.empty() && significant.back() == '0') {
84 significant.remove_suffix(1);
85 }
86
87 out.push_back('.');
88 out.append(significant);
89 return out;
90}
