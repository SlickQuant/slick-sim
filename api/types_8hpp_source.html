72inline std::string to_fixed_string(int_fast64_t value, unsigned min_frac_digits = 0) {
73 const bool negative = value < 0;
74 // Negated through the unsigned domain so the most negative value cannot overflow.
75 const uint64_t magnitude = negative
76 ? (0ULL - static_cast<uint64_t>(value))
77 : static_cast<uint64_t>(value);
78
79 const uint64_t scale = static_cast<uint64_t>(DOUBLE_MULTIPLIER);
80 uint64_t fraction = magnitude % scale;
81
82 std::string out;
83 if (negative) {
84 out.push_back('-');
85 }
86 out += std::to_string(magnitude / scale);
87
88 if (fraction == 0 && min_frac_digits == 0) {
89 return out;
90 }
91
92 char digits[8];
93 for (int i = 7; i >= 0; --i) {
94 digits[i] = static_cast<char>('0' + fraction % 10);
95 fraction /= 10;
96 }
97 std::string_view significant(digits, 8);
98 while (significant.size() > min_frac_digits && significant.back() == '0') {
99 significant.remove_suffix(1);
100 }
101
102 out.push_back('.');
103 out.append(significant);
104 return out;
105}

117inline bool iequals(std::string_view a, std::string_view b) noexcept {
118 if (a.size() != b.size()) {
119 return false;
120 }
121 for (size_t i = 0; i < a.size(); ++i) {
122 char ca = a[i];
123 char cb = b[i];
124 if (ca >= 'a' && ca <= 'z') { ca = static_cast<char>(ca - ('a' - 'A')); }
125 if (cb >= 'a' && cb <= 'z') { cb = static_cast<char>(cb - ('a' - 'A')); }
126 if (ca != cb) {
127 return false;
128 }
129 }
130 return true;
131}
