21 ws_thread_ = std::thread([this]() {

23 setup_routes(app);

25 listen_socket_ = listen_socket;
26 if (listen_socket) {
27 LOG_INFO("{} Publisher started on port {}", to_string(venue_), port_);
28
29 // Get the event loop
30 loop_ = uWS::Loop::get();
31
32 // Create heartbeat timer in the event loop
33 // Use defer() to create timer with proper lambda capture
34 loop_->defer([this]() {
35 // The third argument is ext_size, not a flag: it is the number
36 // of bytes us_timer_ext() hands back. Passing 0 and then
37 // writing a pointer there overruns the timer allocation.
38 heartbeat_timer_ = us_create_timer((struct us_loop_t*)loop_, 0,
39 sizeof(WebsocketMarketDataPublisher*));

42 *((WebsocketMarketDataPublisher**)us_timer_ext(heartbeat_timer_)) = this;
43
44 // Set timer callback that retrieves 'this' from extension data
45 us_timer_set(heartbeat_timer_, [](struct us_timer_t *timer) {
46 // Dereference the pointer stored in extension data
47 auto* self = *((WebsocketMarketDataPublisher**)us_timer_ext(timer));
48 self->check_heartbeats();

50 });

65void WebsocketMarketDataPublisher::stop() {
66 running_.store(false, std::memory_order_release);
67
68 // Close heartbeat timer
69 if (heartbeat_timer_) {
70 us_timer_close(heartbeat_timer_);
71 heartbeat_timer_ = nullptr;
72 }

90void WebsocketMarketDataPublisher::publish_processing() {
91 // Drain up to this many updates before handing the loop back, so socket
92 // writes and the heartbeat timer still get serviced under load.
93 //
94 // Every batch costs one defer, so the limit sets the ratio of scheduling
95 // overhead to work: at 100 a flood re-entered the loop once per 100 updates,
96 // at 1000 once per 1000. The yield still happens often enough that a client
97 // socket is never starved - a batch is bounded by the queue, and the two
98 // snapshot types below cut it short on their own because they are far more
99 // expensive to encode than an incremental update.
100 constexpr int max_batch = 1000;
101 int processed = 0;
102
103 while (processed < max_batch) {
104 // Try to read from the queue (non-blocking)
105 auto [data, size] = md_queue_.read(data_cursor_);
106
107 if (data == nullptr) {
108 // No more data available
109 break;
110 }
111
112 auto *update = reinterpret_cast<MarketDataUpdate*>(data);
113 publish_market_data_update(update);
114
115 if (update->type == MDUpdateType::SUB_RESPONSE ||
116 update->type == MDUpdateType::BOOK_SNAPSHOT) {
117 break; // A whole book went out - yield rather than start another
118 }
119
120 ++processed;
121 }
122
123 // Unconditionally, even when the queue came up empty. This defer is the only
124 // thing that runs publish_processing() again: md_queue_ is a lock-free queue
125 // with no wakeup, and the exchange thread that fills it never touches the
126 // loop. Skipping the reschedule on an empty queue - which looks like an easy
127 // win - stops the publisher permanently the first time it catches up.
128 // schedule_publish_processing() checks the running_ flag.
129 schedule_publish_processing();
130}
