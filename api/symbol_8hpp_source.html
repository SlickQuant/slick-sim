102 void creditTradedQty(Side side, price_t price, qty_t phantom_consumed, time_t event_time) {
103 if (phantom_consumed <= 0 || side > Side::SELL) {
104 return;
105 }
106 auto &credits = traded_qty_credit_[static_cast<size_t>(side)][price];
107 // One credit per timestamp. A single print's sweep of one level arrives as
108 // one summary, but a batch can carry several prints stamped alike.
109 for (auto &credit : credits) {
110 if (credit.event_time == event_time) {
111 credit.qty += phantom_consumed;
112 return;
113 }
114 }
115 credits.emplace_back(phantom_consumed, event_time);
116 }

126 qty_t takeTradedQty(Side side, price_t price, time_t as_of) {
127 if (side > Side::SELL) {
128 return 0;
129 }
130 auto &by_price = traded_qty_credit_[static_cast<size_t>(side)];
131 auto it = by_price.find(price);
132 if (it == by_price.end()) {
133 return 0;
134 }
135
136 auto &credits = it->second;
137 qty_t unreflected = 0;
138 for (auto credit = credits.begin(); credit != credits.end();) {
139 if (as_of < credit->event_time) {
140 // The update predates this trade, so it still reports the
141 // pre-trade quantity at this price.
142 unreflected += credit->qty;
143 ++credit;
144 } else {
145 credit = credits.erase(credit);
146 }
147 }
148 if (credits.empty()) {
149 by_price.erase(it);
150 }
151 return unreflected;
152 }
