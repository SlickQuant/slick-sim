107void HyperliquidRestOrderGateway::handle_exchange(
108 uWS::HttpResponse<false>* res, uWS::HttpRequest* req)
109{
110 auto request_data = std::make_shared<PostRequestData>();
111
112 res->onAborted([request_data]() {
113 LOG_WARN("HyperliquidREST: POST /exchange aborted");
114 });
115
116 res->onData([this, res, req, request_data](std::string_view chunk, bool is_last) mutable {
117 request_data->body_buffer.append(chunk);
118 if (!is_last) return;
119
120 try {
121 json body = json::parse(request_data->body_buffer);
122 std::string user_id = extract_wallet(body, req);
123
124 if (!body.contains("action")) {
125 res->writeStatus("400 Bad Request")
126 ->writeHeader("Content-Type", "application/json")
127 ->end(R"({"status":"err","response":"missing action"})");
128 return;
129 }
130 const auto& action = body["action"];
131 std::string type = action.value("type", "");
132
133 if (type == "order") {
134 process_order_action(res, action, user_id);
135 } else if (type == "cancel") {
136 process_cancel_action(res, action, user_id);
137 } else if (type == "cancelByCloid") {
138 process_cancel_by_cloid_action(res, action, user_id);
139 } else if (type == "batchModify") {
140 process_batch_modify_action(res, action, user_id);
141 } else {
142 res->writeStatus("400 Bad Request")
143 ->writeHeader("Content-Type", "application/json")
144 ->end(std::format(R"({{"status":"err","response":"unsupported action type: {}"}})", type));
145 }
146 } catch (const std::exception& e) {
147 res->writeStatus("400 Bad Request")
148 ->writeHeader("Content-Type", "application/json")
149 ->end(std::format(R"({{"status":"err","response":"parse error: {}"}})", e.what()));
150 }
151 });
