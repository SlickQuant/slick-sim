299void CoinbaseWebsocketOrderGateway::process_all_responses() {
300 // Process up to 100 responses per call to avoid blocking event loop too long
301 constexpr int max_batch = 100;
302 int processed = 0;
303
304 while (processed < max_batch) {
305 // Try to read from the queue (non-blocking)
306 auto [data, size] = response_queue_.read(response_read_index_);
307
308 if (data == nullptr) {
309 // No more data available
310 break;
311 }
312
313 // Process the order response
314 auto& response = *data;
315
316 // Record it before anything else. This has to happen whether or not the
317 // owner is connected: a client that subscribes later still expects its
318 // earlier orders in the snapshot, and this is the only pass over the stream.
319 orders_.apply(response);
320
321 // Extract user_id from the order response
322 std::string user_id = response.user_id;
323
324 if (user_id.empty()) {
325 LOG_WARN("Order response has empty user_id, skipping");
326 processed++;
327 continue;
328 }
329
330 // Find the WebSocket connection for this user (no lock needed - single threaded)
331 auto it = clients_.find(user_id);
332 if (it != clients_.end()) {
333 auto* client_ws = it->second;
334 // Serialize order response to JSON
335 handle_order_response(response, client_ws);
336
337 LOG_INFO("Sent order response {} to user_id={}, client_order_id={}, order_status={}",
338 to_string(response.exec_type), user_id, response.client_order_id, to_string(response.order_status));
339 } else {
340 LOG_WARN("No WebSocket connection found for user_id={}", user_id);
341 }
342
343 processed++;
344 }
345
346 // Reschedule ourselves to run again on next event loop iteration
347 // schedule_response_processing() checks running_ flag
348 schedule_response_processing();
349}

351void CoinbaseWebsocketOrderGateway::handle_order_response(const OrderResponse &response, wsT *ws) {
352 // Serialize order response to JSON
353 if (response.order_status == OrderStatus::PENDING_NEW) {
354 // Don't send pending new updates to client - wait for final status update
355 return;
356 }
357 auto filled_value = to_price_double(response.avg_fill_price) * to_qty_double(response.cum_qty);
358 auto fees = filled_value * fee_rate_;
359 auto *user_data = ws->getUserData();
360 json response_json = {
361 {"channel", "user"},
362 {"client_id", ""},
363 {"timestamp", format_timestamp_iso8601(9)},
364 {"sequence_num", user_data->seq_num++},
365 {"events", json::array({
366 json::object({
367 {"type", "update"},
368 {"orders", json::array({
369 json::object({
370 {"avg_price", std::to_string(to_price_double(response.avg_fill_price))},
371 {"client_order_id", std::string(response.client_order_id)},
372 {"completion_percentage", std::to_string(response.qty > 0 ? (to_qty_double(response.cum_qty) / response.qty) * 100 : 0.0)},
373 {"contract_expiry_type", "UNKNOWN_CONTRACT_EXPIRY_TYPE"},
374 {"cumulative_quantity", std::to_string(to_qty_double(response.cum_qty))},
375 {"filled_value", std::to_string(filled_value)},
376 {"leaves_quantity", std::to_string(to_qty_double(response.leaves_qty))},
377 {"limit_price", std::to_string(to_price_double(response.price))},
378 {"number_of_fills", std::to_string(response.last_qty > 0 ? response.cum_qty / response.last_qty : 0)},
379 {"order_id", std::string(response.order_id)},
380 {"order_side", response.side == Side::BUY ? "BUY" : "SELL"},
381 {"order_type", to_string(response.order_type)},
382 {"post_only", response.post_only ? "true" : "false"},
383 {"product_id", std::string(response.symbol)},
384 {"reject_reason", to_string(response.reject_reason)},
385 {"status", to_string(to_coinbase_order_status(response.order_status))},
386 {"time_in_force", to_string(response.time_in_force)},
387 {"total_fees", std::to_string(fees)},
388 {"total_value_after_fees", std::to_string(filled_value - fees)},
389 {"trigger_status", "INVALID_ORDER_TYPE"},
390 {"creation_time", format_timestamp_iso8601(response.creation_time, 9)},
391 {"end_time", "0001-01-01T00:00:00Z"},
392 {"start_time", "0001-01-01T00:00:00Z"}
393 })
394 })}
395 })
396 })}

403void CoinbaseWebsocketOrderGateway::check_heartbeats() {
404 auto now = std::chrono::steady_clock::now();
405 std::vector<std::string> disconnected_clients;
406
407 // No lock needed - single threaded event loop
408 for (auto it = clients_.begin(); it != clients_.end();) {
409 auto* ws = it->second;
410 auto* user_data = ws->getUserData();
411
412 // Check if client has responded to pings
413 auto time_since_pong = std::chrono::duration_cast<std::chrono::seconds>(
414 now - user_data->last_pong);
415
416 if (time_since_pong > pong_timeout_) {
417 // Client hasn't responded to pings, close connection
418 LOG_WARN("Client {} failed heartbeat check (last_pong: {}s ago), disconnecting",
419 user_data->user_id, time_since_pong.count());
420
421 disconnected_clients.push_back(it->first);
422 it = clients_.erase(it);
423 ws->close(); // Close the connection
424 } else {
425 // Send ping to client (only if authenticated)
426 if (user_data->subscribed_heartbeat) {
427 // Send a text-based ping message (Coinbase style)
428 json event = {
429 {"current_time", utils::format_heatbeat_timestamp()},
430 {"heartbeat_counter", user_data->heartbeat_count++}
431 };
432 json ping_msg = {
433 {"channel", "heartbeats"},
434 {"client_id", ""},
435 {"timestamp", utils::format_timestamp_iso8601(9)},
436 {"sequence_num", user_data->seq_num++},
437 {"events", {event}}
438 };
439 ws->send(ping_msg.dump(), uWS::OpCode::TEXT);
440 }
441 // Also send protocol-level ping
442 ws->send("", uWS::OpCode::PING);
443 LOG_TRACE("Sent heartbeat to client {}", user_data->user_id);
444 ++it;
445 }
446 }
447
448 if (!disconnected_clients.empty()) {
449 LOG_INFO("Disconnected {} stale clients", disconnected_clients.size());
450 }
