13bool is_multiple_of(double value, double increment) {
14 if (increment < 1e-12) return true;
15 double ratio = value / increment;
16 return std::abs(ratio - std::round(ratio)) < 1e-6;
17}

49std::tuple<bool, std::string, std::string> CoinbaseRestOrderGateway::authenticate(uWS::HttpRequest *req) {
50 // Get the Authorization header
51 std::string_view auth_header = req->getHeader("authorization");
52
53 // For Coinbase, it's typically: "Bearer <JWT_TOKEN>"
54 if (auth_header.empty()) {
55 auth_header = req->getHeader("Authorization");
56 if (auth_header.empty()) {
57 return std::make_tuple(false, "", R"({"error":"UNAUTHORIZED","message":"Authorization header required"})");
58 }
59 }
60
61 // Extract the token (remove "Bearer " prefix)
62 std::string token;
63 if (auth_header.starts_with("Bearer ")) {
64 token = std::string(auth_header.substr(7));
65 } else {
66 return std::make_tuple(false, "", R"({"error":"UNAUTHORIZED","message":"Invalid auth token"})");
67 }
68
69 try {
70 auto decoded = jwt::decode<jwt::traits::nlohmann_json>(token);
71
72 // Extract 'sub' claim
73 std::string sub = decoded.get_subject();
74 std::string user_id = sub.substr(sub.rfind('/') + 1);
75 return std::make_tuple(true, user_id, "");
76
77 } catch (const std::exception& /* e */) {
78 return std::make_tuple(false, "", R"({"error":"UNAUTHORIZED","message":"Invalid token"})");
79 }

82CoinbaseRestOrderGateway::OrderValidationResult
83CoinbaseRestOrderGateway::validate_order(
84 const std::string& product_id, double price_d, double qty_d, OrderType order_type) const
85{
86 auto it = product_by_id_.find(product_id);
87 if (it == product_by_id_.end()) {
88 return {false, "INVALID_PRODUCT_ID", "Unknown product: " + product_id};
89 }
90
91 auto pj = json::parse(it->second);
92 auto get_d = [&](const char* key) -> double {
93 if (!pj.contains(key)) return 0.0;
94 auto& v = pj[key];
95 return v.is_string() ? std::stod(v.get<std::string>()) : v.get<double>();
96 };
97
98 const double price_inc = get_d("price_increment");
99 const double base_inc = get_d("base_increment");
100 const double base_min = get_d("base_min_size");
101
102 if (order_type != OrderType::MARKET && price_inc > 1e-12) {
103 if (!is_multiple_of(price_d, price_inc)) {
104 return {false, "INVALID_LIMIT_PRICE_POST_ONLY",
105 "price " + std::to_string(price_d) +
106 " is not a multiple of tick size " + std::to_string(price_inc)};
107 }
108 }
109
110 if (base_min > 1e-12 && qty_d < base_min - 1e-9) {
111 return {false, "INVALID_ORDER_CONFIG",
112 "qty " + std::to_string(qty_d) +
113 " is below minimum order size " + std::to_string(base_min)};
114 }
115
116 if (base_inc > 1e-12 && !is_multiple_of(qty_d, base_inc)) {
117 return {false, "INVALID_ORDER_CONFIG",
118 "qty " + std::to_string(qty_d) +
119 " is not a multiple of base increment " + std::to_string(base_inc)};
120 }
121
122 return {true, {}, {}};
123}

473void CoinbaseRestOrderGateway::handle_batch_cancel(uWS::HttpResponse<false>* res, uWS::HttpRequest* req) {
474 auto [authenticated, user_id, error_response] = authenticate(req);
475 if (!authenticated) {
476 res->writeStatus("401 Unauthorized")
477 ->writeHeader("Content-Type", "application/json")
478 ->end(error_response);
479 return;
480 }
481
482 auto request_data = std::make_shared<PostRequestData>();
483 request_data->user_id = std::move(user_id);
484
485 res->onAborted([request_data]() {
486 LOG_WARN("HTTP connection aborted during POST /api/v3/brokerage/orders/batch_cancel");
487 });
488
489 res->onData([this, res, request_data](std::string_view chunk, bool is_last) mutable {
490 request_data->body_buffer.append(chunk);
491 if (!is_last) return;
492
493 try {
494 json request_body = json::parse(request_data->body_buffer);
495 auto& order_ids_json = request_body["order_ids"];
496 std::string user_id = request_data->user_id;
497
498 if (!order_ids_json.is_array() || order_ids_json.empty()) {
499 res->writeStatus("400 Bad Request")
500 ->writeHeader("Content-Type", "application/json")
501 ->end(R"({"error":"INVALID_ARGUMENT","message":"order_ids must be a non-empty array"})");
502 return;
503 }
504
505 struct CancelEntry {
506 std::string order_id;
507 bool submitted;
508 OrdRejectReason reject_reason = OrdRejectReason::NONE;
509 };
510 std::vector<CancelEntry> entries;
511 entries.reserve(order_ids_json.size());
512
513 // Process cancels one at a time so we can correlate reject responses.
514 // The engine looks up by client_order_id first if non-empty, so we must
515 // leave client_order_id empty and look up by order_id only.
516 // Successful cancels don't generate responses (sendOrderCancelPending is
517 // a no-op), so no response within the timeout window means success.
518 constexpr int timeout_ms = 200;
519 constexpr int poll_interval_us = 100; // used inside per-order wait loop
520
521 for (auto& oid_json : order_ids_json) {
522 std::string order_id = oid_json.get<std::string>();
523
524 auto sym_it = order_id_to_symbol_.find(order_id);
525 if (sym_it == order_id_to_symbol_.end()) {
526 LOG_WARN("batch_cancel: unknown order_id={}", order_id);
527 entries.push_back({order_id, false});
528 continue;
529 }
530 const std::string& symbol = sym_it->second;
531
532 // Capture response index before publishing so we don't miss a fast reject
533 uint64_t response_read_index = response_queue_.initial_reading_index();
534
535 auto request_index = request_queue_.reserve();
536 auto* request = request_queue_[request_index];
537 request->time_stamp = utils::get_current_time_ns();
538 std::memset(request->symbol, 0, sizeof(request->symbol));
539 std::memcpy(request->symbol, symbol.c_str(), std::min(sizeof(request->symbol), symbol.size()));
540 request->msg_type = MessageType::ORDER_CANCEL_REQUEST;
541
542 std::memset(&request->cancel_order, 0, sizeof(request->cancel_order));
543 std::memcpy(request->cancel_order.user_id, user_id.c_str(),
544 std::min(sizeof(request->cancel_order.user_id), user_id.size()));
545 // Leave client_order_id empty — engine falls through to findOrderByOrderId
546 std::memcpy(request->cancel_order.order_id, order_id.c_str(),
547 std::min(sizeof(request->cancel_order.order_id), order_id.size()));
548
549 request_queue_.publish(request_index);
550 LOG_INFO("Submitted cancel request: user={}, order_id={}, symbol={}", user_id, order_id, symbol);
551
552 // Wait briefly for a reject. If none arrives, the cancel succeeded.
553 OrdRejectReason reject_reason = OrdRejectReason::NONE;
554 auto deadline = std::chrono::steady_clock::now() + std::chrono::milliseconds(timeout_ms);
555 while (std::chrono::steady_clock::now() < deadline) {
556 auto [data, size] = response_queue_.read(response_read_index);
557 if (data != nullptr) {
558 if (std::string(data->user_id) == user_id && data->request_time == request->time_stamp) {
559 if (data->order_status == OrderStatus::REJECTED) {
560 reject_reason = data->reject_reason;
561 }
562 break;
563 }
564 }
565 std::this_thread::sleep_for(std::chrono::microseconds(poll_interval_us));
566 }
567
568 entries.push_back({order_id, true, reject_reason});
569 }
570
571 // Build results
572 json results = json::array();
573 for (auto& entry : entries) {
574 if (!entry.submitted) {
575 results.push_back({
576 {"success", false},
577 {"failure_reason", "UNKNOWN_ORDER"},
578 {"order_id", entry.order_id}
579 });
580 continue;
581 }
582 if (entry.reject_reason != OrdRejectReason::NONE) {
583 results.push_back({
584 {"success", false},
585 {"failure_reason", to_string(entry.reject_reason)},
586 {"order_id", entry.order_id}
587 });
588 } else {
589 results.push_back({
590 {"success", true},
591 {"failure_reason", ""},
592 {"order_id", entry.order_id}
593 });
594 order_id_to_symbol_.erase(entry.order_id);
595 }

637 auto [authenticated, user_id, error_response] = authenticate(req);
638 if (!authenticated) {
639 res->writeStatus("401 Unauthorized")
640 ->writeHeader("Content-Type", "application/json")
641 ->end(error_response);
642 return;
643 }
644
645 auto request_data = std::make_shared<PostRequestData>();
646 request_data->user_id = std::move(user_id);
647
648 res->onAborted([request_data]() {
649 LOG_WARN("HTTP connection aborted during POST /api/v3/brokerage/orders/edit");
650 });
651
652 res->onData([this, res, request_data](std::string_view chunk, bool is_last) mutable {
653 request_data->body_buffer.append(chunk);
654 if (!is_last) return;
655
656 try {
657 json request_body = json::parse(request_data->body_buffer);
658 std::string order_id = request_body["order_id"];
659 std::string user_id = request_data->user_id;
660
661 auto sym_it = order_id_to_symbol_.find(order_id);
662 if (sym_it == order_id_to_symbol_.end()) {
663 res->writeStatus("400 Bad Request")
664 ->writeHeader("Content-Type", "application/json")
665 ->end(json({
666 {"success", false},
667 {"errors", json::array({{{"edit_failure_reason", "UNKNOWN_ORDER"}}})}
668 }).dump());
669 LOG_WARN("edit_order: unknown order_id={}", order_id);
670 return;
671 }
672 const std::string& symbol = sym_it->second;
673
674 price_t new_price = NULL_PRICE;
675 qty_t new_qty = 0;
676 if (request_body.contains("price")) {
677 new_price = to_price_t(std::stod(request_body["price"].get<std::string>()));
678 }
679 if (request_body.contains("size")) {
680 new_qty = to_qty_t(std::stod(request_body["size"].get<std::string>()));
681 }
682
683 // Validate modified price/qty against product specs before queuing
684 if (new_price != NULL_PRICE || new_qty != 0) {
685 const OrderType ot = (new_price != NULL_PRICE) ? OrderType::LIMIT : OrderType::MARKET;
686 const double price_d = (new_price != NULL_PRICE) ? to_price_double(new_price) : 0.0;
687 const double qty_d = (new_qty != 0) ? to_qty_double(new_qty) : 0.0;
688 auto vr = validate_order(symbol, price_d, qty_d, ot);
689 if (!vr.valid) {
690 res->writeStatus("400 Bad Request")
691 ->writeHeader("Content-Type", "application/json")
692 ->end(json({
693 {"success", false},
694 {"errors", json::array({{{"edit_failure_reason", vr.error_type}}})}
695 }).dump());
696 LOG_ERROR("Edit order rejected (validation): user={}, order_id={}, reason={}",
697 user_id, order_id, vr.error_details);
698 return;
699 }
700 }
701
702 uint64_t response_read_index = response_queue_.initial_reading_index();
703
704 auto request_index = request_queue_.reserve();
705 auto* request = request_queue_[request_index];
706 request->time_stamp = utils::get_current_time_ns();
707 std::memset(request->symbol, 0, sizeof(request->symbol));
708 std::memcpy(request->symbol, symbol.c_str(), std::min(sizeof(request->symbol), symbol.size()));
709 request->msg_type = MessageType::ORDER_REPLACE_REQUEST;

714 std::memcpy(request->modify_order.order_id, order_id.c_str(),
715 std::min(sizeof(request->modify_order.order_id), order_id.size()));
716 request->modify_order.new_price = new_price;
717 request->modify_order.new_qty = new_qty;
718
719 request_queue_.publish(request_index);
720 LOG_INFO("Published edit request: user={}, order_id={}, symbol={}", user_id, order_id, symbol);
721
722 // Wait briefly for a reject. No response within timeout means success.
723 constexpr int timeout_ms = 200;
724 constexpr int poll_interval_us = 100;
725 OrdRejectReason reject_reason = OrdRejectReason::UNKNOWN;
726 auto deadline = std::chrono::steady_clock::now() + std::chrono::milliseconds(timeout_ms);
727
728 std::string client_order_id;
729 while (std::chrono::steady_clock::now() < deadline) {
730 auto [data, size] = response_queue_.read(response_read_index);
731 if (data != nullptr) {
732 if (std::string(data->user_id) == user_id && data->request_time == request->time_stamp) {
733 if (data->order_status == OrderStatus::REJECTED) {
734 reject_reason = data->reject_reason;
735 }
736 else {
737 reject_reason = OrdRejectReason::NONE;
738 }
739 client_order_id = data->client_order_id;
740 break;
741 }
742 }
743 std::this_thread::sleep_for(std::chrono::microseconds(poll_interval_us));
744 }
745
746 if (reject_reason != OrdRejectReason::NONE) {
747 res->writeStatus("400 Bad Request")
748 ->writeHeader("Content-Type", "application/json")
749 ->end(json({
750 {"success", false},
751 {"errors", json::array({{{"edit_failure_reason", to_string(reject_reason)}}})}
752 }).dump());
753 LOG_ERROR("Edit order rejected: user={}, order_id={}, reason={}", user_id, order_id, to_string(reject_reason));
754 } else {
755 res->writeStatus("200 OK")
756 ->writeHeader("Content-Type", "application/json")
757 ->end(json({
758 {"success", true},
759 {"errors", json::array()}
760 }).dump());
761 LOG_INFO("Edit order succeeded: user={}, order_id={}, client_order_id={}", user_id, order_id, client_order_id);
762 }
763
764 } catch (const json::parse_error& e) {
765 res->writeStatus("400 Bad Request")
766 ->writeHeader("Content-Type", "application/json")
767 ->end(json({
768 {"error", "INVALID_REQUEST"},
769 {"message", std::string("Invalid JSON: ") + e.what()}
770 }).dump());
771 LOG_ERROR("JSON parse error in POST /orders/edit: {}", e.what());
772 } catch (const json::exception& e) {

774 ->writeHeader("Content-Type", "application/json")
775 ->end(json({
776 {"error", "INVALID_ARGUMENT"},
777 {"message", std::string("Missing or invalid field: ") + e.what()}
