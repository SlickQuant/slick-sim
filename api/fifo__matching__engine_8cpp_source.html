15// Helper: Check if incoming order would self-match with book order
16// Returns true if orders are from same client
17inline bool isSelfMatch(const Order* incoming_order, const Order* book_order) {
18 if (book_order == nullptr) {
19 return false; // Book order not found, assume different client
20 }
21
22 // SMP only applies when both orders have valid client_ids
23 if (incoming_order->client_id == -1 || book_order->client_id == -1) {
24 return false;
25 }
26
27 return incoming_order->client_id == book_order->client_id;
28}

30// Helper: Handle self-match according to Symbol's configured SMP mode
31// Returns true if matching should continue, false if order should be rejected
32inline bool handleSelfMatch(MatchingEngine& engine, const Order* incoming_order, Order* book_order,
33 OrderBook& book, slick::sim::time_t request_time,
34 uint64_t event_time, uint64_t seq_num,
35 SelfMatchPreventionMode smp_mode) {
36 switch (smp_mode) {
37 case SelfMatchPreventionMode::NONE:
38 // Allow self-matching
39 return true;
40
41 case SelfMatchPreventionMode::CANCEL_RESTING:
42 // Cancel the resting book order and continue matching. This is a real
43 // cancel, so it takes the same three steps as Symbol::cancelOrder:
44 // remove it from the book, tell its owner, and return it to the pool.
45 // Deleting it from the book alone would leave the client believing the
46 // order was still live and leak the pooled Order.
47 LOG_INFO("{} SMP: Canceling resting order {} (client_id={})",
48 incoming_order->symbol.view(), book_order->order_id.view(), incoming_order->client_id);
49 book.deleteOrder(book_order, event_time, seq_num, false);
50 engine.publishOrderCancel(book_order, request_time);
51 book.freeOrder(book_order);
52 return true; // Continue to next order
53
54 case SelfMatchPreventionMode::CANCEL_NEWEST:
55 // Reject the incoming order. Reached only as a backstop — the
56 // pre-check in match() decides this before anything is mutated.
57 LOG_INFO("{} SMP: Rejecting new order {} due to self-match with resting order {} (client_id={})",
58 incoming_order->symbol.view(), incoming_order->id, book_order->order_id.view(), incoming_order->client_id);
59 return false; // Signal to reject order
60
61 default:
62 return true;
63 }
64}

66// Pre-check for CANCEL_NEWEST: would this order reach a resting order of its own?
67//
68// This has to be answered before the order is touched. An amendment mutates
69// price/quantity and reports REPLACED on its way into the matching loop, and a
70// new order is acked there, so discovering the self-match mid-loop would mean
71// telling the client the order was replaced or live and then rejecting it.
72// Only orders the incoming quantity could actually reach are considered.
73template<Side SIDE>
74bool wouldSelfMatch(const Order* order, price_t order_price, qty_t order_qty, OrderBook& book) {
75 qty_t remaining = order_qty;
76
77 const auto& levels = book.getLevelsL3(static_cast<slick::orderbook::Side>(SIDE));
78 for (const auto& [level_price, level_data] : levels) {
79 if (order_price != NULL_PRICE && !isPriceBetterOrEqual<SIDE>(level_price, order_price)) [[unlikely]] {
80 break; // Reached price levels we can't match against
81 }
82
83 for (const auto& book_order : level_data.orders) {
84 if (remaining <= 0) [[unlikely]] {
85 return false;
86 }
87 if (isSelfMatch(order, book.findOrder(book_order.order_id))) [[unlikely]] {
88 return true;
89 }
90 remaining -= std::min<qty_t>(book_order.quantity, remaining);
91 }
92 }
93 return false;
94}

96// Pre-check for Fill-Or-Kill: Determine if entire order can be filled
97// Returns true if sufficient liquidity exists (considering SMP)
98template<Side SIDE>
99bool canFillCompletely(const Order* order, price_t order_price, qty_t order_qty, OrderBook& book,
100 SelfMatchPreventionMode smp_mode) {
101 qty_t remaining_check_qty = order_qty;
102
103 // Self-match handling is the only reason to resolve a book order to its full
104 // Order, and that is a hash lookup per resting order. With SMP off nothing
105 // downstream needs it, so the whole scan stays inside the level's own array.
106 const bool check_self_match = smp_mode != SelfMatchPreventionMode::NONE;
107
108 // Get all levels on the opposite side of the book
109 const auto& levels = book.getLevelsL3(static_cast<slick::orderbook::Side>(SIDE));
110
111 // Iterate through price levels in order (best to worst)
112 for (const auto& [level_price, level_data] : levels) {
113 // Check price compatibility - stop if price is worse than our limit
114 if (order_price != NULL_PRICE && !isPriceBetterOrEqual<SIDE>(level_price, order_price)) [[unlikely]] {
115 break; // Reached price levels we can't match against
116 }
117
118 // Check each order in this level
119 for (const auto& book_order : level_data.orders) {
120 // Handle self-matches based on SMP mode (exceptional case)
121 if (check_self_match && isSelfMatch(order, book.findOrder(book_order.order_id))) [[unlikely]] {
122 if (smp_mode == SelfMatchPreventionMode::CANCEL_NEWEST) [[unlikely]] {
123 // CANCEL_NEWEST: Order would be rejected on first self-match
124 return false;
125 }
126 // CANCEL_RESTING: This resting order would be canceled, so skip it.
127 // NONE never gets here - the quantity stays countable because the
128 // order really would trade against itself.
129 LOG_INFO("{} FOK check: Skipping self-match with resting order {} due to SMP (CANCEL_RESTING mode)",
130 order->symbol.view(), book_order.order_id);
131 continue;
132 }
133
134 // Count available quantity (normal path)
135 remaining_check_qty -= std::min<qty_t>(book_order.quantity, remaining_check_qty);
136
137 if (remaining_check_qty == 0) [[likely]] {
138 return true; // Found enough liquidity
139 }
140 }
141 }
142
143 // Every path that finds enough liquidity has already returned, so reaching
144 // here means the book ran out - except for a zero-quantity order, which is
145 // trivially fillable.
146 return remaining_check_qty == 0;
147}

149template<Side SIDE>
150std::tuple<OrdRejectReason, std::vector<TradeSummaryInfo>> match(MatchingEngine& engine, Order* order, price_t order_price, qty_t order_qty, OrderBook &book, slick::sim::time_t request_time, slick::sim::time_t event_time, uint64_t seq_num, SelfMatchPreventionMode smp_mode) {
151 std::vector<TradeSummaryInfo> trade_summaries;
152
153 // STEP 0: CANCEL_NEWEST is decided before the order is acked, mutated or
154 // reported on, so a rejected order leaves no trace: no NEW, no REPLACED, no
155 // partial fills, and the stored Order keeps its original price and quantity.
156 // The caller publishes the single terminal REJECTED.
157 if (smp_mode == SelfMatchPreventionMode::CANCEL_NEWEST) {
158 // Scan only as deep as this order will really trade. `order_qty` is the
159 // order's *total* quantity, which for a partially filled amendment is
160 // more than its remaining leaves — scanning that far could reach one of
161 // the owner's resting orders the amendment would never touch and reject
162 // a perfectly good fill. Mirror the leaves the matching loop is about to
163 // compute below, without mutating anything yet.
164 qty_t effective_qty = order->leaves_quantity;
165 if (order->status == OrderStatus::PENDING_REPLACE) {
166 effective_qty += order_qty - order->quantity;
167 if (effective_qty < 0) {
168 effective_qty = 0;
169 }
170 }
171
172 if (effective_qty > 0 && wouldSelfMatch<SIDE>(order, order_price, effective_qty, book)) {
173 LOG_INFO("{} Order {} rejected due to SMP (CANCEL_NEWEST mode)", order->symbol.view(), order->id);
174 return std::make_tuple(OrdRejectReason::SMP, trade_summaries);
175 }
176 }
177
178 if (order->status == OrderStatus::PENDING_NEW) {
179 engine.publishOrderAck(order, request_time);
180 }
181
182 // STEP 1: Fill-Or-Kill pre-validation
183 if (order->time_in_force == TimeInForce::FILL_OR_KILL) [[unlikely]] {
184 if (!canFillCompletely<SIDE>(order, order_price, order_qty, book, smp_mode)) [[unlikely]] {
185 engine.publishOrderCancel(order, request_time); // Can't fill the order fully, immediate cancel
186 LOG_INFO("{} FOK order {} cannot be completely filled, canceling", order->symbol.view(), order->id);
187 return std::make_tuple(OrdRejectReason::FOK_CANNOT_FILL, trade_summaries);
188 }
189 }
190
191 if (order->status == OrderStatus::PENDING_REPLACE) [[unlikely]] {
192 order->quantity = order_qty;
193 order->price = order_price;
194 if (order->cum_quantity >= order_qty) [[unlikely]] {
195 order->leaves_quantity = 0;
196 } else {
197 order->leaves_quantity = order_qty - order->cum_quantity;
198 }
199 engine.publishOrderModify(order, order_price, order_qty, request_time);
200 }
201
202 if (order->leaves_quantity == 0) [[unlikely]] {
203 LOG_INFO("{} Order {} has zero leaves quantity after modification, canceling", order->symbol.view(), order->id);
204 engine.publishOrderCancel(order, request_time);
205 return std::make_tuple(OrdRejectReason::NONE, trade_summaries);
206 }
207
208 // STEP 2: Matching loop with SMP
209 auto last_fill_price = NULL_PRICE;
210 order_qty = order->leaves_quantity; // Use leaves quantity for matching
211
212 while (order_qty > 0) {
213 auto best_level = book.getBestLevel<static_cast<slick::orderbook::Side>(SIDE)>();
214
215 // Check if we should continue matching at this level
216 if (!best_level || (order_price != NULL_PRICE && !isPriceBetterOrEqual<SIDE>(best_level->price, order_price))) {
217 break;
218 }
219
220 // Refresh orders reference each time (level may have been modified)
221 auto& orders = best_level->orders;
222 assert(!orders.empty() && "Price level has no orders");
223
224 auto order_it = orders.begin();
225 auto &book_order = *order_it;
226
227 // NEW: Self-Match Prevention check (exceptional case)
228 auto* full_book_order = book.findOrder(book_order.order_id);
229 if (smp_mode != SelfMatchPreventionMode::NONE && isSelfMatch(order, full_book_order)) [[unlikely]] {
230 // Handle self-match according to Symbol's SMP mode
231 bool should_continue = handleSelfMatch(engine, order, full_book_order, book,
232 request_time, event_time, seq_num, smp_mode);
233
234 if (!should_continue) [[unlikely]] {
235 // CANCEL_NEWEST backstop: STEP 0 should already have rejected this.
236 // No cancel is published — the caller emits the one terminal REJECTED.
237 return std::make_tuple(OrdRejectReason::SMP, trade_summaries);
238 }
239
240 // CANCEL_RESTING mode: resting order was deleted, continue to next order
241 continue;
242 }
243
244 auto price = best_level->price;
245 auto book_order_id = book_order.order_id;
246 qty_t trade_qty = std::min<qty_t>(order_qty, book_order.quantity);
247
248 LOG_INFO("{} Matching order {} with book order {} for qty {} at price {}", order->symbol.view(), order->id, book_order_id, trade_qty, price);
249 executeOrder(order, price, trade_qty, event_time);
250 order_qty -= trade_qty;
251
252 engine.publishOrderExecution(order);
253
254 book.executeOrder(book_order_id, trade_qty, event_time, seq_num, order_qty == 0);
255
256 // A resting order the simulator does not own is phantom liquidity mirroring
257 // the real market. Tracked separately so the exchange can tell how much of
258 // this trade the venue's own level update will also account for.
259 auto phantom_qty = full_book_order ? qty_t{0} : trade_qty;
260
261 // A trade has two sides, and when the resting side is one of the
262 // simulator's own orders it belongs to a client who has to hear about it.
263 // book.executeOrder() above booked the fill onto it, but OrderBook holds no
264 // reference to the engine so it cannot publish, and on a complete fill it
265 // only drops the order from the lookup maps - leaving the pooled Order
266 // unreturned. Both are this loop's job, exactly as the market-data replay
267 // path below does them. Read `full_book_order` before freeing it.
268 if (full_book_order) [[unlikely]] {
269 engine.publishOrderExecution(full_book_order);
270 if (full_book_order->leaves_quantity == 0) {
271 // Erases whatever the maps still hold and returns it to the pool.
272 book.deleteOrder(full_book_order);
273 }
274 }
275
276 if (last_fill_price != price) [[unlikely]] {
277 // Create trade summary
278 TradeSummaryInfo summary;
279 summary.timestamp = event_time;
280 summary.trade_id = MatchingEngine::nextTradeId();
281 summary.aggressor_side = opposite_side<SIDE>(); // SIDE is the resting side, so aggressor is the opposite
282 summary.price = price;
283 summary.qty = trade_qty;
284 summary.phantom_qty = phantom_qty;
285 summary.num_orders = 2;
286 summary.Trades.emplace_back(Trade{order->id, trade_qty});
287 summary.Trades.emplace_back(Trade{book_order_id, trade_qty});
288
289 trade_summaries.emplace_back(std::move(summary));
290 last_fill_price = price;
291 } else {
292 // Update last trade summary for this price level
293 auto &last_summary = trade_summaries.back();
294 last_summary.num_orders += 1;
295 last_summary.qty += trade_qty;
296 last_summary.phantom_qty += phantom_qty;
297 last_summary.Trades.front().qty += trade_qty; // Update aggressor trade qty
298 last_summary.Trades.emplace_back(Trade{book_order_id, trade_qty});
299 }
300 }
301
302 if (order->time_in_force == TimeInForce::IMMEDIATE_OR_CANCEL && order_qty > 0) {
303 // FOK: Cancel any remaining quantity if we couldn't fill the entire order
304 engine.publishOrderCancel(order, request_time);
305 LOG_INFO("{} Order {} has TIF=IOC and was not completely filled, canceling remaining qty {}", order->symbol.view(), order->id, order_qty);
306 }
307
308 return std::make_tuple(OrdRejectReason::NONE, trade_summaries);
309}
