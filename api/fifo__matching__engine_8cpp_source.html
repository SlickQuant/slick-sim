15// Helper: Check if incoming order would self-match with book order
16// Returns true if orders are from same client
17inline bool isSelfMatch(const Order* incoming_order, const Order* book_order) {
18 if (book_order == nullptr) {
19 return false; // Book order not found, assume different client
20 }
21
22 // SMP only applies when both orders have valid client_ids
23 if (incoming_order->client_id == -1 || book_order->client_id == -1) {
24 return false;
25 }
26
27 return incoming_order->client_id == book_order->client_id;
28}

30// Helper: Handle self-match according to Symbol's configured SMP mode
31// Returns true if matching should continue, false if order should be rejected
32inline bool handleSelfMatch(const Order* incoming_order, Order* book_order,
33 OrderBook& book, uint64_t event_time, uint64_t seq_num,
34 SelfMatchPreventionMode smp_mode) {
35 switch (smp_mode) {
36 case SelfMatchPreventionMode::NONE:
37 // Allow self-matching
38 return true;
39
40 case SelfMatchPreventionMode::CANCEL_RESTING:
41 // Cancel the resting book order and continue matching
42 LOG_INFO("{} SMP: Canceling resting order {} (client_id={})",
43 incoming_order->symbol, book_order->order_id, incoming_order->client_id);
44 book.deleteOrder(book_order, event_time, seq_num, false);
45 return true; // Continue to next order
46
47 case SelfMatchPreventionMode::CANCEL_NEWEST:
48 // Reject the incoming order
49 LOG_INFO("{} SMP: Rejecting new order {} due to self-match with resting order {} (client_id={})",
50 incoming_order->symbol, incoming_order->id, book_order->order_id, incoming_order->client_id);
51 return false; // Signal to reject order
52
53 default:
54 return true;
55 }
56}

58// Pre-check for Fill-Or-Kill: Determine if entire order can be filled
59// Returns true if sufficient liquidity exists (considering SMP)
60template<Side SIDE>
61bool canFillCompletely(const Order* order, price_t order_price, qty_t order_qty, OrderBook& book,
62 SelfMatchPreventionMode smp_mode) {
63 qty_t available_qty = 0;
64 qty_t remaining_check_qty = order_qty;
65
66 // Get all levels on the opposite side of the book
67 const auto& levels = book.getLevelsL3(static_cast<slick::orderbook::Side>(SIDE));
68
69 // Iterate through price levels in order (best to worst)
70 for (const auto& [level_price, level_data] : levels) {
71 // Check price compatibility - stop if price is worse than our limit
72 if (order_price != NULL_PRICE && !isPriceBetterOrEqual<SIDE>(level_price, order_price)) {
73 break; // Reached price levels we can't match against
74 }
75
76 // Check each order in this level
77 for (const auto& book_order : level_data.orders) {
78 // Look up full order to check client_id
79 auto* full_order = book.findOrder(book_order.order_id);
80
81 // Handle self-matches based on SMP mode
82 if (isSelfMatch(order, full_order)) {
83 if (smp_mode == SelfMatchPreventionMode::CANCEL_NEWEST) {
84 // CANCEL_NEWEST: Order would be rejected on first self-match
85 return false;
86 }
87 else if (smp_mode == SelfMatchPreventionMode::CANCEL_RESTING) {
88 // CANCEL_RESTING: This resting order would be canceled, so skip it
89 LOG_INFO("{} FOK check: Skipping self-match with resting order {} due to SMP (CANCEL_RESTING mode)",
90 order->symbol, book_order.order_id);
91 continue;
92 }
93 }
94
95 // Count available quantity
96 qty_t usable_qty = std::min<qty_t>(book_order.quantity, remaining_check_qty);
97 available_qty += usable_qty;
98 remaining_check_qty -= usable_qty;
99
100 if (remaining_check_qty == 0) {
101 return true; // Found enough liquidity
102 }
103 }
104
105 // If we've filled the order completely, we're done
106 if (remaining_check_qty == 0) {
107 break;
108 }
109 }
110
111 return available_qty >= order_qty;
112}
