66 // std::priority_queue pops the *greatest* element, so every comparison here is
67 // inverted: `a < b` must mean "a is processed after b".
68 struct EventCompare {
69 bool operator()(const Event& a, const Event& b) const {
70 if (a.event_time != b.event_time) {
71 // Earliest event time first.
72 return a.event_time > b.event_time;
73 }
74 if (a.type != b.type) {
75 // A trade and the level update reflecting it share a timestamp but
76 // arrive on independent channels, so sequence_id (arrival order)
77 // is arbitrary between them. Always apply the trade first: the
78 // level update is then measured against a book that has already
79 // absorbed it, instead of reducing the same quantity twice.
80 return a.type != EventType::TRADE;
81 }
82 // Same instant, same kind: pop the *highest* sequence_id first. That
83 // reads backwards next to the inverted event_time comparison above, but
84 // it is load-bearing — do not "fix" it to match. sequence_id is assigned
85 // as each event is enqueued, and the two enqueue loops deliberately walk
86 // their batch in opposite directions so that this ordering replays both
87 // chronologically:
88 //
89 // onLevel2Updates iterates the batch in REVERSE, so array index 0
90 // gets the highest id and replays first — Coinbase
91 // sends level updates oldest-first.
92 // onMarketTrades iterates FORWARD, so the last array element gets
93 // the highest id and replays first — Coinbase sends
94 // market_trades newest-first.
95 //
96 // Invert it and both batches replay backwards: levels settle on a stale
97 // quantity, and trades match against the book and take their public
98 // trade ids in reverse.
99 return a.sequence_id < b.sequence_id;
100 }
101 };
