45[[nodiscard]] inline std::vector<std::string> resolve_pcap_files(const std::string &root, const std::string &address,
46 uint16_t port, uint64_t begin_ns,
47 uint64_t end_ns) {
48 namespace fs = std::filesystem;
49 constexpr int64_t DAY_S = 86400;
50 std::vector<std::string> files;
51 const std::string group = std::format("{}:{}", address, port);
52 const int64_t first_day = static_cast<int64_t>(begin_ns / 1'000'000'000ull) / DAY_S - 1;
53 const int64_t last_day = static_cast<int64_t>(end_ns / 1'000'000'000ull) / DAY_S + 1;
54 for (int64_t day = first_day; day <= last_day; ++day) {
55 const std::time_t t = static_cast<std::time_t>(day * DAY_S);
56 std::tm tm{};
57 gmtime_r(&t, &tm);
58 const fs::path dir = fs::path(root) / std::format("{:04}-{:02}-{:02}", tm.tm_year + 1900, tm.tm_mon + 1, tm.tm_mday) / group;
59 for (const char *ext : {".pcap.zst", ".pcap.gz", ".pcap"}) {
60 const fs::path file = dir / (group + ext);
61 std::error_code ec;
62 if (fs::is_regular_file(file, ec)) {
63 files.push_back(file.string());
64 break;
65 }
66 }
67 }
68 return files;
69}

154 bool peek(Item &item) {
155 if (cursor_ >= current_.entries.size() && !fetch()) {
156 return false;
157 }
158 const Entry &entry = current_.entries[cursor_];
159 item.ts_ns = entry.ts_ns;

216 bool fetch() {
217 Batch *next = batches_.read();
218 if (!next) {
219 return false;
220 }
221 // Swap, so the slot keeps the spent batch's buffers for the reader to fill again.
222 std::swap(current_, *next);
223 cursor_ = 0;
224 batches_.release(); // the slot is the reader's again
225 return !current_.entries.empty();
226 }

236 void trim_files(Stream &s) {
237 if (s.files.size() < 2) {
238 return;
239 }
240 std::vector<std::optional<uint64_t>> starts;
241 for (const auto &file : s.files) {
242 try {
243 PcapReader reader(file);
244 PcapPacket packet;
245 starts.push_back(reader.next(packet) ? std::optional<uint64_t>(packet.ts_ns) : std::nullopt);
246 } catch (const std::exception &) {
247 starts.push_back(std::nullopt);
248 }
249 }
250 std::vector<std::string> kept;
251 for (size_t i = 0; i < s.files.size(); ++i) {
252 const bool covered_by_next = i + 1 < s.files.size() && starts[i + 1] && *starts[i + 1] <= begin_ns_;
253 const bool after_window = starts[i] && *starts[i] > end_ns_;
254 if (!covered_by_next && !after_window) {
255 kept.push_back(std::move(s.files[i]));
256 }
257 }
258 // Everything skipped: leave the newest, so the error (or the lack of packets) is the real one.
259 if (kept.empty()) {
260 kept.push_back(s.files.back());
261 }
262 s.files = std::move(kept);
263 }

266 void advance(Stream &s) {
267 s.has_head = false;
268 PcapPacket packet;
269 for (;;) {
270 // Positioning decompresses to `begin`, which can take minutes: let stop() cut it short.
271 if (stopping_.load(std::memory_order_relaxed)) {
272 s.reader.reset();
273 s.exhausted = true;
274 return;
275 }
276 if (!s.reader) {
277 if (s.next_file >= s.files.size()) {
278 s.exhausted = true;
279 return;
280 }
281 s.reader = std::make_unique<PcapReader>(s.files[s.next_file++]);
282 s.truncated_counted = 0;
283 s.overlap_until = s.last_ts;
284 }
285 const bool got = s.reader->next(packet);
286 if (const uint64_t truncated = s.reader->truncated(); truncated != s.truncated_counted) {
287 truncated_frames_.fetch_add(truncated - s.truncated_counted, std::memory_order_relaxed);
288 s.truncated_counted = truncated;
289 }
290 if (!got) {
291 if (!s.reader->error().empty()) {
292 throw std::runtime_error(std::format("{}: {}", s.reader->path(), s.reader->error()));
293 }
294 s.reader.reset();
295 continue;
296 }
297 if (packet.dst_addr != s.addr || packet.dst_port != s.config.port) {
298 continue;
299 }
300 if (packet.ts_ns < begin_ns_ || packet.ts_ns <= s.overlap_until) {
301 continue;
302 }
303 s.overlap_until = 0;
304 if (packet.ts_ns > end_ns_) {
305 s.reader.reset();
306 s.exhausted = true;
307 return;
308 }
309 s.head.assign(packet.data, packet.data + packet.size);
310 s.head_ts = s.last_ts = packet.ts_ns;
311 s.has_head = true;
312 return;
313 }
314 }

318 Batch *acquire_batch(uint64_t &index) {
319 while ((index = batches_.try_reserve()) == slick::spsc_queue<Batch>::invalid_index) {
320 if (stopping_.load(std::memory_order_relaxed)) {
321 return nullptr;
322 }
323 std::this_thread::sleep_for(std::chrono::microseconds(200));
324 }
325 Batch *batch = batches_[index];
326 batch->entries.clear();
327 batch->bytes.clear();
328 return batch;
329 }

331 void run() {
332 try {
333 std::vector<Stream> streams(lines_.size());
334 for (size_t i = 0; i < lines_.size(); ++i) {
335 streams[i].config = lines_[i];
336 streams[i].addr = parse_ipv4(lines_[i].address);
337 streams[i].files = lines_[i].paths.empty()
338 ? resolve_pcap_files(root_, lines_[i].address, lines_[i].port, begin_ns_, end_ns_)
339 : lines_[i].paths;
340 if (streams[i].files.empty()) {
341 throw std::runtime_error(std::format("replay {}: no capture files for {}:{} in the window",
342 name_, lines_[i].address, lines_[i].port));
343 }
344 }
345 // Skipping to `begin` decompresses everything before it, per line and
346 // independently, so do the lines at once.
347 std::vector<std::exception_ptr> failures(streams.size());
348 std::vector<std::thread> positioning;
349 for (size_t i = 0; i < streams.size(); ++i) {
350 positioning.emplace_back([&, i] {
351 try {
352 trim_files(streams[i]);
353 advance(streams[i]);
354 } catch (...) {
355 failures[i] = std::current_exception();
356 }
357 });
358 }
359 for (auto &t : positioning) {
360 t.join();
361 }
362 for (auto &failure : failures) {
363 if (failure) {
364 std::rethrow_exception(failure);
365 }
366 }
367 positioned_.store(true, std::memory_order_release);
368
369 // Earliest head first. Equal timestamps keep the line order, so A and B copies
370 // of one packet replay together, A first.
371 auto later = [&](size_t a, size_t b) {
372 return streams[a].head_ts != streams[b].head_ts ? streams[a].head_ts > streams[b].head_ts : a > b;
373 };
374 std::priority_queue<size_t, std::vector<size_t>, decltype(later)> heads(later);
375 for (size_t i = 0; i < streams.size(); ++i) {
376 if (streams[i].has_head) {
377 heads.push(i);
378 }
379 }
380 Batch *batch = nullptr; // the ring slot being filled
381 uint64_t slot = 0;
382 while (!heads.empty()) {
383 if (stopping_.load(std::memory_order_relaxed)) {
384 break;
385 }
386 if (!batch && !(batch = acquire_batch(slot))) {
387 break;
388 }
389 const size_t i = heads.top();
390 heads.pop();
391 Stream &s = streams[i];
392 batch->entries.push_back({s.head_ts, static_cast<uint32_t>(batch->bytes.size()),
393 static_cast<uint32_t>(s.head.size()), s.config.line});
394 batch->bytes.insert(batch->bytes.end(), s.head.begin(), s.head.end());
395 read_ts_ns_.store(s.head_ts, std::memory_order_relaxed);
396 advance(s);
397 if (s.has_head) {
398 heads.push(i);
399 }
400 if (batch->entries.size() >= BATCH_PACKETS || batch->bytes.size() >= BATCH_BYTES) {
401 batches_.publish(slot);
402 batch = nullptr;
403 }
404 }
405 if (batch && !batch->entries.empty() && !stopping_.load(std::memory_order_relaxed)) {
406 batches_.publish(slot);
407 }
408 } catch (const std::exception &e) {
409 error_ = e.what();
410 failed_.store(true, std::memory_order_release); // publishes error_, which is never written again
411 }
412 // An error before the files were positioned must not leave a waiter hanging.
413 positioned_.store(true, std::memory_order_release);
414 done_.store(true, std::memory_order_release);
415 }

452 void anchor(uint64_t capture_ns, uint64_t wall_ns) {

454 anchor_wall_ns_ = wall_ns;
455 anchored_ = true;

458 [[nodiscard]] bool anchored() const { return anchored_; }
459
461 [[nodiscard]] uint64_t due_ns(uint64_t capture_ns) const {
462 if (speed_ == 0 || capture_ns <= anchor_capture_ns_) {
463 return anchor_wall_ns_;
464 }
465 return anchor_wall_ns_ + static_cast<uint64_t>(static_cast<double>(capture_ns - anchor_capture_ns_) / speed_);
466 }
467
469 [[nodiscard]] bool is_due(uint64_t capture_ns, uint64_t now_ns) const {
470 return speed_ == 0 || now_ns >= due_ns(capture_ns);
471 }
472
476 [[nodiscard]] uint64_t rebase_ns(uint64_t capture_ns, uint64_t now_ns) const {
477 if (speed_ == 0) {
478 return now_ns;
479 }
480 return std::min(due_ns(capture_ns), now_ns);
481 }

486 uint64_t anchor_wall_ns_ = 0;
