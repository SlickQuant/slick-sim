36void CliServer::add_endpoints(const std::string &group, const std::vector<CliEndpoint *> &endpoints) {
37 auto route = [this](std::string_view target, size_t index) {
38 if (target.empty()) {
39 return;
40 }
41 if (target.size() > sizeof(CliRequestMessage::target)) {
42 LOG_WARN("CLI: target {} is longer than a request can name", target);
43 }
44 auto it = targets_.find(target);
45 if (it == targets_.end()) {
46 it = targets_.emplace(std::string(target), std::vector<size_t>{}).first;
47 }
48 if (std::find(it->second.begin(), it->second.end(), index) == it->second.end()) {
49 it->second.push_back(index);
50 }
51 };
52 for (CliEndpoint *endpoint : endpoints) {
53 auto found = std::find(endpoints_.begin(), endpoints_.end(), endpoint);
54 const auto index = static_cast<size_t>(found - endpoints_.begin());
55 if (found == endpoints_.end()) {
56 endpoints_.push_back(endpoint);
57 }
58 route(group, index);
59 const std::string_view name = endpoint->name();
60 for (size_t slash = name.find('/'); slash != std::string_view::npos; slash = name.find('/', slash + 1)) {
61 route(name.substr(0, slash), index);
62 }
63 route(name, index);
64 }
65}

151void CliServer::dispatch(int client_id) {
152 auto it = clients_.find(client_id);
153 if (it == clients_.end()) {
154 return;
155 }
156 auto &client = it->second;
157 while (client.ticket == 0 && !client.waiting.empty()) {
158 const CliRequestMessage request = client.waiting.front();
159 client.waiting.pop_front();
160 const auto target = field(request.target, sizeof(request.target));
161 const auto arg = field(request.arg, sizeof(request.arg));
162 // Copies: a packed field cannot bind to the logger's references.
163 const uint32_t request_id = request.request_id;
164 const int flag = request.flag;
165 LOG_INFO("CLI: connection {} request {}: {} {} {}{}{}", client_id, request_id, target,
166 to_string(request.command), flag, arg.empty() ? "" : " ", arg);
167
168 if (request.command == CliCommandType::LIST_TARGETS) {
169 send_reply(client_id, request.request_id, CliStatus::OK, list_targets());
170 continue;
171 }
172 if (!known(request.command)) {
173 send_reply(client_id, request.request_id, CliStatus::FAILED,
174 std::format("unknown command {}", static_cast<int>(request.command)));
175 continue;
176 }
177 auto route = targets_.find(target);
178 if (route == targets_.end()) {
179 send_reply(client_id, request.request_id, CliStatus::FAILED,
180 std::format("unknown target '{}'; ask for the targets", target));
181 continue;
182 }
183 const auto &indices = route->second;
184 if (std::any_of(indices.begin(), indices.end(), [this](size_t i) { return endpoints_[i]->busy(); })) {
185 send_reply(client_id, request.request_id, CliStatus::FAILED, std::format("'{}' is busy", target));
186 continue;
187 }
188
189 const uint64_t ticket = next_ticket_++;
190 Command command{ticket, request.command, request.flag, {}};
191 std::memcpy(command.arg, request.arg, sizeof(command.arg));
192 Pending pending{client_id, request.request_id, {}, indices.size(),
193 std::chrono::steady_clock::now() + reply_timeout_};
194 pending.answers.reserve(indices.size());
195 for (size_t index : indices) {
196 pending.answers.push_back(Answer{index, false, CliStatus::FAILED, {}});
197 // None was busy, and a target names each endpoint once.
198 [[maybe_unused]] const bool posted = endpoints_[index]->post(command);
199 assert(posted);
200 }
201 pending_.emplace(ticket, std::move(pending));
202 pending_count_.store(pending_.size(), std::memory_order_release);
203 client.ticket = ticket;
204 }
205}

207void CliServer::on_reply(size_t endpoint, uint64_t ticket, CliStatus status, std::string_view text) {
208 auto it = pending_.find(ticket);
209 if (it == pending_.end()) {
210 return; // Timed out; the client has had its answer.
211 }
212 for (auto &answer : it->second.answers) {
213 if (answer.endpoint == endpoint && !answer.answered) {
214 answer.answered = true;
215 answer.status = status;
216 answer.text.assign(text);
217 if (--it->second.remaining == 0) {
218 finish(ticket);
219 }
220 return;
221 }
222 }

225void CliServer::finish(uint64_t ticket) {
226 auto it = pending_.find(ticket);
227 if (it == pending_.end()) {
228 return;
229 }
230 Pending pending = std::move(it->second);
231 pending_.erase(it);
232
233 CliStatus status = CliStatus::OK;
234 std::string text;
235 for (const auto &answer : pending.answers) {
236 const auto &name = endpoints_[answer.endpoint]->name();
237 if (!answer.answered) {
238 status = CliStatus::FAILED;
239 text += std::format("{}: no answer within {} ms\n", name, reply_timeout_.count());
240 LOG_WARN("CLI: {} did not answer request {} within {} ms", name, pending.request_id,
241 reply_timeout_.count());
242 continue;
243 }
244 if (answer.status != CliStatus::OK) {
245 status = CliStatus::FAILED;
246 }
247 std::string_view rest = answer.text;
248 do {
249 const size_t eol = rest.find('\n');
250 text += name;
251 text += ": ";
252 text += rest.substr(0, eol);
253 text += '\n';
254 rest = eol == std::string_view::npos ? std::string_view{} : rest.substr(eol + 1);
255 } while (!rest.empty());
256 }
257 if (!text.empty()) {
258 text.pop_back();
259 }
260
261 if (pending.client_id >= 0) {
262 send_reply(pending.client_id, pending.request_id, status, text);
263 if (auto client = clients_.find(pending.client_id); client != clients_.end()) {
264 client->second.ticket = 0;
265 }
266 }
267 pending_count_.store(pending_.size(), std::memory_order_release);
268 if (pending.client_id >= 0) {
269 dispatch(pending.client_id);
270 }
271}
