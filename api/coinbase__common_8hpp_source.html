22constexpr RoutingEffect routing_effect(OrderStatus status) {
23 switch (status) {
24 case OrderStatus::FILLED:
25 case OrderStatus::CANCELED:
26 case OrderStatus::EXPIRED:
27 case OrderStatus::DONE_FOR_DAY:
28 return RoutingEffect::Forget;
29 case OrderStatus::REJECTED:
30 return RoutingEffect::Ignore;
31 case OrderStatus::NEW:
32 case OrderStatus::PARTIALLY_FILLED:
33 case OrderStatus::REPLACED:
34 case OrderStatus::PENDING_CANCEL:
35 case OrderStatus::PENDING_NEW:

37 case OrderStatus::STOPPED:
38 case OrderStatus::SUSPENDED:
39 case OrderStatus::CALCULATED:
40 case OrderStatus::ACCEPTED_FOR_BIDDING:
41 return RoutingEffect::Learn;

43 return RoutingEffect::Learn;
