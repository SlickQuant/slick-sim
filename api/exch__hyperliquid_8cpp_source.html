149// Full order-book snapshot (documented l2Book shape, and also what
150// Hyperliquid's undocumented l2 channel sends as the very first message
151// after subscribing). The whole book is authoritative here, so it's safe —
152// and necessary, to re-validate crossing — to clear all phantom liquidity
153// and rebuild every level from scratch. Also resets the "r" index-resolution
154// mirror (l2_ref_levels_) from this snapshot's own (already-ordered) levels.
155void HyperliquidExchange::processL2Snapshot(const nlohmann::json& data) {
156 if (!data.contains("coin") || !data.contains("levels")) return;
157
158 const std::string coin = data["coin"].get<std::string>();
159
160 auto *symbol = sym_mgr.getSymbol(coin);
161 if (!symbol) {
162 symbol = addSymbol(coin);
163 }
164
165 // levels[0] = bids, levels[1] = asks
166 const auto& levels_arr = data["levels"];
167 if (levels_arr.size() < 2) return;
168
169 uint64_t event_time_ms = data.value("time", uint64_t{0});
170 uint64_t event_time_ns = event_time_ms * utils::ONE_MILLISECOND_NS;
171
172 symbol->order_book_->clearMDOrders();
173
174 auto &ref = l2_ref_levels_[symbol];
175 ref[0].clear();
176 ref[1].clear();
177
178 auto process_side = [&](const nlohmann::json& side_levels, Side side, std::vector<price_t>& ref_side) {
179 auto book_side = to_book_side(side);
180 for (const auto& level : side_levels) {
181 double px = std::stod(level["px"].get<std::string>());
182 double sz = std::stod(level["sz"].get<std::string>());
183 auto price = to_price_t(px);
184 auto qty = to_qty_t(sz);
185 if (qty <= 0) [[unlikely]] continue;
186 auto order_id = utils::nextOrderId();
187 if (symbol->matching_engine_) {
188 auto trade_summaries = symbol->matching_engine_->match(
189 side, order_id, price, qty, *symbol->order_book_, event_time_ns, 0);
190 for (const auto& summary : trade_summaries) {
191 publishTradeSummary(symbol, summary);
192 }
193 }
194 if (qty > 0) {
195 symbol->order_book_->addOrder(order_id, book_side, price, qty, event_time_ns, 0, 0, true);
196 }
197 // Snapshot levels already arrive best-first, so appending
198 // preserves the correct order.
199 ref_side.push_back(price);
200 }
201 };
202
203 process_side(levels_arr[0], Side::BUY, ref[0]);
204 process_side(levels_arr[1], Side::SELL, ref[1]);
205
206 symbol->order_book_->setLastUpdate(event_time_ns, 0);
207
208 finalizeL2Event(symbol);
209
210 symbol->md_level_update_cache_.clear();
211 symbol->md_order_update_cache_.clear();
212 LOG_TRACE("Hyperliquid l2 snapshot applied: {}", coin);
213}

215// Incremental diff from Hyperliquid's undocumented l2 channel: "l" gives the
216// new absolute quantity at changed/added price levels; "r" gives indices
217// into the *previous* per-side ordered price array (l2_ref_levels_) for
218// levels that were removed entirely. Only the touched levels are mutated —
219// everything else in the book is left exactly as it was.
220void HyperliquidExchange::processL2Diff(const nlohmann::json& data) {
221 if (!data.contains("c")) return;
222
223 const auto coin = data["c"].get<std::string_view>();
224
225 auto *symbol = sym_mgr.getSymbol(coin);
226 if (!symbol) {
227 symbol = addSymbol(coin);
228 }
229
230 // levels[0] = bids, levels[1] = asks
231 const auto& levels_arr = data["l"];
232 if (levels_arr.size() < 2) return;
233
234 uint64_t event_time_ms = data.value("t", uint64_t{0});
235 uint64_t event_time_ns = event_time_ms * utils::ONE_MILLISECOND_NS;
236
237 auto &ref = l2_ref_levels_[symbol];
238
239 auto apply_removals = [&](const nlohmann::json& indices, Side side, std::vector<price_t>& ref_side) {
240 // Resolve every index against the array as it stood *before* this
241 // event's mutations (removing while iterating would shift later
242 // indices), then apply.
243 std::vector<price_t> removed_prices;
244 removed_prices.reserve(indices.size());
245 for (const auto& idx_json : indices) {
246 auto idx = idx_json.get<size_t>();
247 if (idx < ref_side.size()) {
248 removed_prices.push_back(ref_side[idx]);
249 } else {
250 LOG_WARN("Hyperliquid l2 diff: removed-level index {} out of range (have {}) for {}",
251 idx, ref_side.size(), coin);
252 }
253 }
254 for (price_t price : removed_prices) {
255 applyPhantomLevelUpdate(symbol, side, price, 0, event_time_ns);
256 if (auto it = std::find(ref_side.begin(), ref_side.end(), price); it != ref_side.end()) {
257 ref_side.erase(it);
258 }
259 }
260 };
261
262 if (const auto& removed = data.value("r", nlohmann::json::array()); removed.is_array() && removed.size() == 2) {
263 apply_removals(removed[0], Side::BUY, ref[0]);
264 apply_removals(removed[1], Side::SELL, ref[1]);
265 }
266
267 auto apply_changes = [&](const nlohmann::json& side_levels, Side side, std::vector<price_t>& ref_side) {
268 for (const auto& level : side_levels) {
269 double px = std::stod(level["p"].get<std::string>());
270 double sz = std::stod(level["s"].get<std::string>());
271 auto price = to_price_t(px);
272 auto qty = to_qty_t(sz);
273
274 applyPhantomLevelUpdate(symbol, side, price, qty, event_time_ns);
275
276 auto it = std::lower_bound(ref_side.begin(), ref_side.end(), price,
277 [side](price_t a, price_t b) { return utils::isPriceBetter(side, a, b); });
278 if (it == ref_side.end() || *it != price) {
279 ref_side.insert(it, price);
280 }
281 }
282 };
283
284 apply_changes(levels_arr[0], Side::BUY, ref[0]);
285 apply_changes(levels_arr[1], Side::SELL, ref[1]);
286
287 symbol->order_book_->setLastUpdate(event_time_ns, 0);
288
289 finalizeL2Event(symbol);
290
291 symbol->md_level_update_cache_.clear();
292 symbol->md_order_update_cache_.clear();
293 LOG_TRACE("Hyperliquid l2 diff applied: {}", coin);
294}

296// Sets the phantom (real-market-mirroring) liquidity at (side, price) to
297// exactly target_qty, without disturbing any other price level or any order
298// belonging to the simulator's own user (tracked separately via
299// Symbol::findOrder — phantom orders are never registered there). Mirrors
300// the established incremental-update pattern in
301// CoinbaseExchange::dispatchEvent (exch_coinbase.cpp).
302void HyperliquidExchange::applyPhantomLevelUpdate(
303 Symbol* symbol, Side side, price_t price, qty_t target_qty, uint64_t event_time_ns)
304{
305 auto book_side = to_book_side(side);
306 auto &book = *symbol->order_book_;
307 auto [level, index] = book.getLevel(book_side, price);
308
309 if (!level) {
310 // Brand-new price level — the full target quantity is the delta
311 // from an implicit zero, and this price hasn't been checked for
312 // crossing the opposing side yet.
313 if (target_qty <= 0) return;
314 qty_t qty = target_qty;
315 auto order_id = utils::nextOrderId();
316 if (symbol->matching_engine_) {
317 auto trade_summaries = symbol->matching_engine_->match(
318 side, order_id, price, qty, book, event_time_ns, 0);
319 for (const auto& summary : trade_summaries) {
320 publishTradeSummary(symbol, summary);
321 }
322 }
323 if (qty > 0) {
324 book.addOrder(order_id, book_side, price, qty, event_time_ns, 0, 0, true);
325 }
326 return;
327 }
328
329 // Existing level: already known non-crossing, so no need to re-match —
330 // just adjust the phantom-only quantity (getMDLevelQty excludes the
331 // user's own resting quantity at this price) toward the target.
332 reconcilePhantomQty(symbol, side, price, target_qty, event_time_ns, 0, true);
333}

335// Shared tail for both processL2Snapshot and processL2Diff: dispatches the
336// first (SUB_RESPONSE) or routine (BOOK_SNAPSHOT) downstream update for the
337// L2_BOOK/L2 channels. See the design note above pending_md_subscription_.
338void HyperliquidExchange::finalizeL2Event(Symbol* symbol) {
339 // L2_BOOK and L2 (undocumented, compressed-diff variant) are independent
340 // downstream channels over the same book data — a symbol can have a
341 // pending subscriber on one while already having an active subscriber on
342 // the other, so each channel's first-snapshot delivery is tracked
343 // separately.
344 bool l2book_was_pending = false;
345 bool l2_was_pending = false;
346
347 auto &pending_l2book = pending_md_subscription_[static_cast<size_t>(HyperliquidChannel::L2_BOOK)];
348 if (auto it = pending_l2book.find(symbol); it != pending_l2book.end()) {
349 symbol->order_book_->populateL2SubscriptionResponse(
350 md_queue_, static_cast<uint8_t>(HyperliquidChannel::L2_BOOK));
351 pending_l2book.erase(it);
352 l2book_was_pending = true;
353 }
354
355 auto &pending_l2 = pending_md_subscription_[static_cast<size_t>(HyperliquidChannel::L2)];
356 if (auto it = pending_l2.find(symbol); it != pending_l2.end()) {
357 symbol->order_book_->populateL2SubscriptionResponse(
358 md_queue_, static_cast<uint8_t>(HyperliquidChannel::L2));
359 pending_l2.erase(it);
360 l2_was_pending = true;
361 }
362
363 if (!l2book_was_pending || !l2_was_pending) {
364 // Routine update — the publisher filters/formats per actual subscribers.
365 symbol->order_book_->populateL2Snapshot(md_queue_);
366 }
367}

369void HyperliquidExchange::processTradesEvent(const nlohmann::json& data) {
370 if (!data.is_array() || data.empty()) return;
371
372 const std::string coin = data[0].value("coin", "");
373 if (coin.empty()) return;
374
375 auto *symbol = sym_mgr.getSymbol(coin);
376 if (!symbol) return;
377
378 auto &book = *symbol->order_book_;
379
380 for (const auto& trade : data) {
381 std::string side_str = trade.value("side", "B");
382 Side side = (side_str == "B") ? Side::BUY : Side::SELL;
383 double px = std::stod(trade["px"].get<std::string>());
384 double sz = std::stod(trade["sz"].get<std::string>());
385 uint64_t time_ms = trade.value("time", uint64_t{0});
386 uint64_t event_time_ns = time_ms * utils::ONE_MILLISECOND_NS;
387
388 auto price = to_price_t(px);
389 auto book_side = to_book_side(side);
390 auto order_id = utils::nextOrderId();
391
392 // The venue's print is replayed as an incoming aggressor order: it takes
393 // liquidity from the book exactly as the real taker did, filling any
394 // simulator order that was ahead of the phantom liquidity it swept. Only
395 // the resulting fills are published - the print itself is never relayed.
396 qty_t qty = to_qty_t(sz);
397 if (!symbol->matching_engine_) {
398 continue;
399 }
400 auto trade_summaries = symbol->matching_engine_->match(
401 side, order_id, price, qty, book, event_time_ns, 0);
402
403 if (qty > 0) {
404 // Treated as an aggressor order, so the unfilled remainder rests. It
405 // cannot cross: a remainder exists precisely because nothing was
406 // still crossable at this limit.
407 book.addOrder(order_id, book_side, price, qty, event_time_ns, 0, 0, true);
408 }
409
410 for (const auto& summary : trade_summaries) {
411 // The venue will report the same reduction again in its level update
412 // for this trade; credit it so that update does not remove it twice.
413 symbol->creditTradedQty(opposite_side(side), summary.price,
414 summary.phantom_qty, event_time_ns);
415 publishTradeSummary(symbol, summary);
416 }
417 }
418
419 // The l2Book/l2 channels are snapshot-driven here, so republish the book to
420 // show the liquidity these trades consumed.
421 book.populateL2Snapshot(md_queue_);
422 symbol->md_level_update_cache_.clear();
423 symbol->md_order_update_cache_.clear();
424}

426void HyperliquidExchange::handleMdSubscription(const Request &request) {
427 const auto &msg = request.md_subscription;
428 std::string coin = request.symbol;
429
430 auto *symbol = sym_mgr.getSymbol(coin);
431 bool is_new = false;
432 if (!symbol) {
433 symbol = addSymbol(coin);
434 is_new = true;
435 }
436
437 symbol->num_subscriptions_.fetch_add(1, std::memory_order_relaxed);
438
439 auto channel = static_cast<HyperliquidChannel>(msg.channel);
440
441 if (use_live_feed_) {
442 if (is_new) {
443 // Subscribe to live feed for this coin
444 auto it_feed = map_coin_feed_.find(coin);
445 if (it_feed == map_coin_feed_.end()) {
446 // No existing feed — use or create one
447 std::shared_ptr<md_feed::HyperliquidLiveWSFeed> live_feed;
448 if (!md_feeds_.empty()) {
449 live_feed = std::dynamic_pointer_cast<md_feed::HyperliquidLiveWSFeed>(md_feeds_.front());
450 }
451 if (!live_feed) {
452 auto new_feed = std::make_shared<md_feed::HyperliquidLiveWSFeed>(this, live_ws_base_url_);
453 md_feeds_.emplace_back(new_feed);
454 new_feed->start();
455 live_feed = new_feed;
456 }
457 live_feed->addCoin(coin);
458 map_coin_feed_.emplace(coin, live_feed);
459 } else {
460 auto live_feed = std::dynamic_pointer_cast<md_feed::HyperliquidLiveWSFeed>(it_feed->second);
461 if (live_feed) {
462 live_feed->addCoin(coin);
463 }
464 }
465 // Mark as pending — snapshot will arrive on the next l2Book WS message.
466 // The trades channel has no upstream snapshot to wait for; it is served
467 // from the tape history below, which also delivers the subscription
468 // confirmation the publisher only emits on a SUB_RESPONSE.
469 if (channel == HyperliquidChannel::TRADES) {
470 publishTradeSubscriptionResponse(symbol, static_cast<uint8_t>(channel));
471 } else {
472 pending_md_subscription_[static_cast<size_t>(channel)].emplace(symbol);
473 }
474 } else {
475 auto &pending = pending_md_subscription_[static_cast<size_t>(channel)];
476 if (pending.find(symbol) == pending.end()) {
477 // Already have a snapshot, send it immediately
478 if (channel == HyperliquidChannel::L2_BOOK || channel == HyperliquidChannel::L2) {
479 symbol->order_book_->populateL2SubscriptionResponse(
480 md_queue_, static_cast<uint8_t>(channel));
481 } else if (channel == HyperliquidChannel::TRADES) {
482 publishTradeSubscriptionResponse(symbol, static_cast<uint8_t>(channel));
483 }
484 }
485 }
486 } else {
487 // No live feed — send snapshot from whatever is in the book
488 if (channel == HyperliquidChannel::L2_BOOK || channel == HyperliquidChannel::L2) {
489 symbol->order_book_->populateL2SubscriptionResponse(
490 md_queue_, static_cast<uint8_t>(channel));
491 } else if (channel == HyperliquidChannel::TRADES) {
492 publishTradeSubscriptionResponse(symbol, static_cast<uint8_t>(channel));
493 }
494 }
