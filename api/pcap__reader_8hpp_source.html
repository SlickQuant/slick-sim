32[[nodiscard]] inline uint32_t parse_ipv4(const std::string &text) {
33 unsigned a = 0, b = 0, c = 0, d = 0;
34 char tail = 0;
35 if (std::sscanf(text.c_str(), "%u.%u.%u.%u%c", &a, &b, &c, &d, &tail) != 4 || a > 255 || b > 255 || c > 255 ||
36 d > 255) {
37 throw std::runtime_error(std::format("'{}' is not an IPv4 address", text));
38 }
39 return (a << 24) | (b << 16) | (c << 8) | d;

80 bool next(PcapPacket &out) {
81 while (pcap_ != nullptr) {
82 pcap_pkthdr *header = nullptr;
83 const u_char *frame = nullptr;
84 const int rc = pcap_next_ex(pcap_, &header, &frame);
85 if (rc == 1) {

88 }

90 }

93 }
94 // PCAP_ERROR_BREAK (-2) is the end of the file. For a decompressed one, that is
95 // only a clean end if the decompressor says so: a truncated or corrupt archive
96 // that stops on a packet boundary reads the same to libpcap.

108 }

126 bool parse(const pcap_pkthdr &header, const u_char *frame, PcapPacket &out) {
127 const uint8_t *p = frame;
128 size_t left = header.caplen;
129 if (link_type_ == DLT_EN10MB) {
130 if (left < 14) {
131 return cut_short();
132 }
133 uint16_t type = static_cast<uint16_t>((p[12] << 8) | p[13]);
134 p += 14;
135 left -= 14;
136 while ((type == 0x8100 || type == 0x88a8) && left >= 4) {
137 type = static_cast<uint16_t>((p[2] << 8) | p[3]);
138 p += 4;
139 left -= 4;
140 }
141 if (type != 0x0800) {
142 return false;
143 }
144 }
145 if (left < 20) {
146 return cut_short();
147 }
148 if ((p[0] >> 4) != 4) {
149 return false;
150 }
151 const size_t ihl = static_cast<size_t>(p[0] & 0x0f) * 4;
152 const size_t total = static_cast<size_t>((p[2] << 8) | p[3]);
153 const uint16_t frag = static_cast<uint16_t>(((p[6] << 8) | p[7]) & 0x3fff); // MF flag + offset
154 // Only whole datagrams: a fragment has no UDP header to read, or no end.
155 if (p[9] != 17 || frag != 0) {
156 return false;
157 }
158 if (ihl < 20 || left < ihl + 8 || total < ihl + 8) {
159 return cut_short();
160 }
161 const uint8_t *udp = p + ihl;
162 const size_t udp_len = static_cast<size_t>((udp[4] << 8) | udp[5]);
163 if (udp_len < 8) {
164 return cut_short();
165 }
166 // A capture can cut a frame short; trust no more than was recorded.
167 size_t payload = udp_len - 8;
168 if (payload > left - ihl - 8) {
169 return cut_short();
170 }
171 // The reader asked libpcap for nanosecond precision, so tv_usec holds nanoseconds.
172 out.ts_ns = static_cast<uint64_t>(header.ts.tv_sec) * 1'000'000'000ull + static_cast<uint64_t>(header.ts.tv_usec);
173 out.dst_addr = (uint32_t(p[16]) << 24) | (uint32_t(p[17]) << 16) | (uint32_t(p[18]) << 8) | p[19];
174 out.dst_port = static_cast<uint16_t>((udp[2] << 8) | udp[3]);
175 out.data = udp + 8;
176 out.size = payload;
177 return true;
178 }

181 FILE *open_stream(const std::string &path) {
182 // Fail with the real reason here; a decompressor handed a missing file only
183 // produces an empty stream, which libpcap reports as a bad capture.
184 if (::access(path.c_str(), R_OK) != 0) {
185 throw std::runtime_error(std::format("{}: {}", path, std::strerror(errno)));
186 }
187 const char *tool = nullptr;
188 if (path.ends_with(".zst")) {
189 tool = "zstd";
190 } else if (path.ends_with(".gz")) {
191 tool = "gzip";
192 }
193 if (tool == nullptr) {
194 FILE *file = std::fopen(path.c_str(), "rb");
195 if (file == nullptr) {
196 throw std::runtime_error(std::format("{}: {}", path, std::strerror(errno)));
197 }
198 return file;
199 }
200 // Spawned directly, not through popen: pcap_close() closes the FILE* itself, and a
201 // popen stream must be pclose()d. The child is reaped by reap() instead.
202 int fds[2];
203 if (::pipe2(fds, O_CLOEXEC) != 0) {
204 throw std::runtime_error(std::format("{}: pipe: {}", path, std::strerror(errno)));
205 }
206 posix_spawn_file_actions_t actions;
207 posix_spawn_file_actions_init(&actions);
208 posix_spawn_file_actions_adddup2(&actions, fds[1], STDOUT_FILENO);
209 posix_spawn_file_actions_addopen(&actions, STDIN_FILENO, "/dev/null", O_RDONLY, 0);
210 const char *argv[] = {tool, "-dc", "--", path.c_str(), nullptr};
211 const int rc = ::posix_spawnp(&child_, tool, &actions, nullptr, const_cast<char *const *>(argv), environ);
212 posix_spawn_file_actions_destroy(&actions);
213 ::close(fds[1]);
214 if (rc != 0) {
215 ::close(fds[0]);
216 child_ = -1;
217 throw std::runtime_error(std::format("{}: cannot run {}: {}", path, tool, std::strerror(rc)));
218 }
219 FILE *file = ::fdopen(fds[0], "rb");
220 if (file == nullptr) {
221 ::close(fds[0]);
222 reap();
223 throw std::runtime_error(std::format("{}: fdopen: {}", path, std::strerror(errno)));
224 }
225 return file;
226 }
