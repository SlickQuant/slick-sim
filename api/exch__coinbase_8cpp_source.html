199 ready = safe_by_time_gap || safe_by_timeout;

374void CoinbaseExchange::populateMDTradesResponse(Symbol* symbol) {
375 uint32_t num_trades = 0;
376 auto it = trade_snapshots_.find(symbol);
377 if (it != trade_snapshots_.end()) {
378 num_trades = static_cast<uint32_t>(it->second.size());
379 }
380
381 auto sz = static_cast<uint32_t>(sizeof(MarketDataUpdate) + sizeof(MDSubscriptionResponse) + sizeof(MDTradeUpdate) + num_trades * sizeof(MDTrade)); // Empty snapshot
382 auto index = md_queue_.reserve(sz);
383 auto *update = reinterpret_cast<MarketDataUpdate*>(md_queue_[index]);
384 memcpy(update->symbol, symbol->symbol_.c_str(), sizeof(update->symbol));
385 update->venue = venue_;
386 update->type = MDUpdateType::SUB_RESPONSE;
387
388 auto *response = reinterpret_cast<MDSubscriptionResponse*>(update->data);
389 response->channel = coinbase::WebSocketChannel::MARKET_TRADES;
390
391 auto *snapshot = reinterpret_cast<MDTradeUpdate*>(response->data);
392 snapshot->num_trades = num_trades;
393 auto *trades = reinterpret_cast<MDTrade*>(snapshot->trades);
394
395 if (it != trade_snapshots_.end()) {
396 // Coinbase trades are in reverse chronological order
397 for (auto i = num_trades - 1; i < num_trades; i--) {
398 const auto *evt = it->second[i];
399 if (evt == nullptr) [[unlikely]] {
400 continue;
401 }
402 trades->event_time = evt->event_time;
403 trades->price = evt->price;
404 trades->qty = evt->qty;
405 trades->side = evt->side;
406 trades++;
407 }
408 }
409 md_queue_.publish(index, sz);
