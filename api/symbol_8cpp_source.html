17 LOG_INFO("{} Adding order {}: oid={}, client_oid={}, price={}, qty={}, tif={}",
18 symbol_.view(), order->id, order->order_id.view(), order->client_order_id.view(),
19 to_price_double(order->price), to_qty_double(order->quantity), to_string(order->time_in_force));
20
21 // Pass Symbol's SMP mode to matching engine
22 auto [reject_reason, trade_summaries] = matching_engine_->match(
23 order, order->price, order->quantity, *order_book_.get(),
24 request_time, order->created_time, 0, smp_mode_
25 );
26
27 // Only add to book if no rejection and remaining quantity exists
28 bool resting = false;
29 if (reject_reason == OrdRejectReason::NONE && order->leaves_quantity > 0) [[likely]] {
30 // Check if TimeInForce allows resting on book
31 bool should_rest = (
32 order->time_in_force == TimeInForce::DAY ||
33 order->time_in_force == TimeInForce::GOOD_TILL_CANCEL ||
34 order->time_in_force == TimeInForce::GOOD_TILL_DATE
35 );
36
37 if (should_rest) [[likely]] {
38 order_book_->addOrder(order, order->created_time);
39 resting = true;
40 } else {
41 // IOC: Remainder is implicitly cancelled (not added to book)
42 // FOK: Will never reach here (would have been rejected in match)
43 LOG_INFO("{} Order {} has TIF={}, not adding to book (leaves_qty={})",
44 symbol_.view(), order->id, static_cast<char>(order->time_in_force), order->leaves_quantity);
45 }
46 }
47
48 if (!resting) [[unlikely]] {
49 // The book is the only thing that takes ownership of an order, so anything
50 // that did not rest has to go back to the pool here: fully filled orders,
51 // rejects (SMP, FOK), and IOC remainders alike. Freeing only on
52 // leaves_quantity == 0 leaked every order that was rejected or cancelled
53 // with quantity outstanding, and a client repeating a rejected order would
54 // drain the pool.
55 order_book_->freeOrder(order);
56 }
57
58 return std::make_tuple(reject_reason, trade_summaries);
59}
