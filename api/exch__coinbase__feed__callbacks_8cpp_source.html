18void CoinbaseExchange::onMarketDataDisconnected(WebSocketClient* client) {
19 LOG_INFO("WebSocket {:p} market data disconnected", (void*)client);
20
21 // Find the feed that owns this client
22 std::shared_ptr<md_feed::MDFeed> disconnected_feed;
23 for (auto& feed : md_feeds_) {
24 auto* live_feed = dynamic_cast<md_feed::CoinbaseLiveWSFeed*>(feed.get());
25 if (live_feed && live_feed->getClient() == client) {
26 disconnected_feed = feed;
27 break;
28 }
29 }
30
31 if (!disconnected_feed) {
32 LOG_WARN("WebSocket {:p} disconnected but no matching feed found", (void*)client);
33 return;
34 }
35
36 // Collect symbols associated with this feed
37 std::vector<std::string> symbols;
38 for (auto& [sym, feed] : map_symbol_feed_) {
39 if (feed == disconnected_feed) {
40 symbols.push_back(sym);
41 }
42 }
43
44 // Remove old feed from tracking
45 md_feeds_.erase(std::remove(md_feeds_.begin(), md_feeds_.end(), disconnected_feed), md_feeds_.end());
46 for (const auto& sym : symbols) {
47 map_symbol_feed_.erase(sym);
48 }
49
50 if (symbols.empty()) {
51 return;
52 }
53
54 // Mark symbols as pending subscription so the snapshot is re-requested on reconnect
55 for (const auto& sym : symbols) {

57 if (symbol) {
58 pending_md_subscription_[coinbase::WebSocketChannel::LEVEL2].emplace(symbol);
59 pending_md_subscription_[coinbase::WebSocketChannel::MARKET_TRADES].emplace(symbol);
60 trade_snapshots_.erase(symbol);
61 }
62 }
63
64 // Create and start a new feed
65 auto new_feed = std::make_shared<md_feed::CoinbaseLiveWSFeed>(ws_mux_, this, symbols);
66 md_feeds_.emplace_back(new_feed);
67 for (const auto& sym : symbols) {
68 map_symbol_feed_.emplace(sym, new_feed);
69 }
70 new_feed->start();
71
72 LOG_INFO("WebSocket {:p} restarted market data feed for {} symbol(s)", (void*)client, symbols.size());
73}

185void CoinbaseExchange::onMarketTradesSnapshot(WebSocketClient* /* client */, uint64_t seq_num, const std::vector<coinbase::MarketTrade>& snapshots) {
186 auto &pending_subscriptions = pending_md_subscription_[coinbase::WebSocketChannel::MARKET_TRADES];
187 if (pending_subscriptions.empty() || snapshots.empty()) {
188 return;
189 }
190 Symbol* symbol = sym_mgr.getSymbol(snapshots[0].product_id);
191 if (!symbol || !pending_subscriptions.contains(symbol)) {
192 return;
193 }
194
195 auto it = trade_snapshots_.find(symbol);
196 if (it == trade_snapshots_.end()) {
197 it = trade_snapshots_.emplace(symbol, utils::RingBuffer<Event>(16)).first;
198 }
199
200 auto &trade_snapshot = it->second;
201
202 auto sz = static_cast<uint32_t>(sizeof(MarketDataUpdate) + sizeof(MDSubscriptionResponse) + sizeof(MDTradeUpdate) + snapshots.size() * sizeof(MDTrade));
203 auto index = md_queue_.reserve(sz);
204 auto *update = reinterpret_cast<MarketDataUpdate*>(md_queue_[index]);
205 memcpy(update->symbol, symbol->symbol_.c_str(), sizeof(update->symbol));
206 update->venue = venue_;
207 update->type = MDUpdateType::SUB_RESPONSE;
208
209 auto *response = reinterpret_cast<MDSubscriptionResponse*>(update->data);
210 response->channel = coinbase::WebSocketChannel::MARKET_TRADES;
211
212 auto *snapshot = reinterpret_cast<MDTradeUpdate*>(response->data);
213 snapshot->num_trades = static_cast<uint32_t>(snapshots.size());
214
215 auto *trades = reinterpret_cast<MDTrade*>(snapshot->trades);
216
217 auto& state = symbol_event_state_[symbol];
218
219 // Coinbase trades are in reverse chronological order
220 for (auto i = snapshots.size() - 1; i >= 0; i--) {
221 const auto& trade = snapshots[i];
222 trades->price = to_price_t(trade.price);
223 trades->qty = to_qty_t(trade.size);
224 trades->side = static_cast<Side>(trade.side);
225
226 // Store in trade history ring buffer
227 Event evt;
228 evt.type = EventType::TRADE;
229 evt.event_time = trade.time;
230 evt.seq_num = seq_num;
231 evt.sequence_id = state.next_sequence_id++;
232 evt.received_time = utils::get_current_time_ns();
233 evt.price = to_price_t(trade.price);
234 evt.qty = to_qty_t(trade.size);
235 evt.side = static_cast<Side>(trade.side);
236 trade_snapshot.emplace(std::move(evt));
237
238 trades++;
239 }
240 for (const auto& trade : snapshots) {
241 trades->price = to_price_t(trade.price);
242 trades->qty = to_qty_t(trade.size);
243 trades->side = static_cast<Side>(trade.side);
244 trades++;
245 }
246 md_queue_.publish(index, sz);
247 pending_subscriptions.erase(symbol);
