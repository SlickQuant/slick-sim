21void CoinbaseExchange::onMarketDataDisconnected(WebSocketClient *client)
22{
23 LOG_INFO("WebSocket {:p} market data disconnected", (void *)client);
24
25 // Find the feed that owns this client
26 std::shared_ptr<md_feed::MDFeed> disconnected_feed;
27 for (auto &feed : md_feeds_)
28 {
29 auto *live_feed = dynamic_cast<md_feed::CoinbaseLiveWSFeed *>(feed.get());
30 if (live_feed && live_feed->getClient() == client)
31 {
32 disconnected_feed = feed;
33 break;
34 }
35 }
36
37 if (!disconnected_feed)
38 {
39 LOG_WARN("WebSocket {:p} disconnected but no matching feed found", (void *)client);
40 return;
41 }
42
43 // Collect symbols associated with this feed
44 std::vector<std::string> symbols;
45 for (auto &[sym, feed] : map_symbol_feed_)
46 {
47 if (feed == disconnected_feed)
48 {
49 symbols.push_back(sym);
50 }
51 }
52
53 // Remove old feed from tracking
54 md_feeds_.erase(std::remove(md_feeds_.begin(), md_feeds_.end(), disconnected_feed), md_feeds_.end());
55 for (const auto &sym : symbols)
56 {
57 map_symbol_feed_.erase(sym);
58 }
59
60 if (symbols.empty())
61 {
62 return;
63 }
64
65 // Mark symbols as pending subscription so the snapshot is re-requested on reconnect
66 for (const auto &sym : symbols)
67 {
68 auto *symbol = sym_mgr.getSymbol(sym);
69 if (symbol)
70 {
71 pending_md_subscription_[coinbase::WebSocketChannel::LEVEL2].emplace(symbol);
72 pending_md_subscription_[coinbase::WebSocketChannel::MARKET_TRADES].emplace(symbol);
73 symbol->clearTradeHistory();
74 }
75 }
76
77 // Create and start a new feed
78 auto new_feed = std::make_shared<md_feed::CoinbaseLiveWSFeed>(ws_mux_, this, symbols);
79 md_feeds_.emplace_back(new_feed);
80 for (const auto &sym : symbols)
81 {
82 map_symbol_feed_.emplace(sym, new_feed);
83 }
84 new_feed->start();
85
86 LOG_INFO("WebSocket {:p} restarted market data feed for {} symbol(s)", (void *)client, symbols.size());
87}

218void CoinbaseExchange::onMarketTradesSnapshot(WebSocketClient * /* client */, uint64_t seq_num, const std::vector<coinbase::MarketTrade> &snapshots)
219{
220 auto &pending_subscriptions = pending_md_subscription_[coinbase::WebSocketChannel::MARKET_TRADES];
221 if (pending_subscriptions.empty() || snapshots.empty())
222 {
223 return;
224 }
225 Symbol *symbol = sym_mgr.getSymbol(snapshots[0].product_id);
226 if (!symbol || !pending_subscriptions.contains(symbol))
227 {
228 return;
229 }
230
231 // These prints happened before we were listening, so they only seed the
232 // history that serves the subscribe snapshot - they are never fed to the
233 // matching engine. Coinbase sends them newest first, so walk in reverse to
234 // store them chronologically.
235 for (auto i = static_cast<int32_t>(snapshots.size()) - 1; i >= 0; i--)
236 {
237 const auto &trade = snapshots[i];
238
239 // Keep the venue's own trade id, and keep the ids this exchange mints
240 // afterwards continuing the venue's sequence rather than colliding with it.
241 uint64_t trade_id = 0;
242 auto [_, ec] = std::from_chars(trade.trade_id.data(), trade.trade_id.data() + trade.trade_id.size(), trade_id);
243 if (ec != std::errc{})
244 {
245 trade_id = nextVenueTradeId();
246 }
247 else if (trade_id >= next_trade_id_)
248 {
249 next_trade_id_ = trade_id + 1;
250 }
251
252 symbol->recordTrade(MDTrade{
253 .trade_id = trade_id,
254 .event_time = trade.time,
255 .seq_num = seq_num,
256 .price = to_price_t(trade.price),
257 .qty = to_qty_t(trade.size),
258 // Never matched into the book, so whether a subscriber's book
259 // snapshot reflects this print depends on its timestamp.
260 .flags = UpdateFlags::F_NOT_IN_BOOK,
261 .side = static_cast<Side>(trade.side)});
262 }
263
264 publishTradeSubscriptionResponse(symbol, coinbase::WebSocketChannel::MARKET_TRADES);
265 pending_subscriptions.erase(symbol);
266}
