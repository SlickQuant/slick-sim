29void ProRataMatchingEngine::allocate(const slick::orderbook::detail::PriceLevelL3 &level, qty_t qty, OrderBook &book, bool resolve) {
30 allocations_.clear();
31 qty_t total = 0;
32 for (const auto &book_order : level.orders) {
33 allocations_.push_back({book_order.order_id, book_order.quantity, 0,
34 resolve ? book.findOrder(book_order.order_id) : nullptr});
35 total += book_order.quantity;
36 }
37
38 if (qty >= total) {
39 // The aggressor takes the whole level, so there is nothing to split.
40 for (auto &a : allocations_) {
41 a.qty = a.size;
42 }
43 return;
44 }
45
46 // From here qty < total, so every step below can place all it is given.
47 qty_t left = qty;
48
49 // 1. The FIFO slice.
50 if (config_.fifo_percent > 0) {
51 qty_t fifo = floorLot(share_of(qty, config_.fifo_percent, 100));
52 left -= fifo;
53 for (auto &a : allocations_) {
54 if (fifo == 0) {
55 break;
56 }
57 a.qty = std::min(fifo, a.size);
58 fifo -= a.qty;
59 }
60 }
61
62 // 2. Pro-rata on what each order still has, against what the level still has.
63 if (left > 0) {
64 const qty_t pro_rata = left;
65 const qty_t base = total - (qty - left);
66 for (auto &a : allocations_) {
67 qty_t share = floorLot(share_of(pro_rata, a.size - a.qty, base));
68 if (share < config_.min_allocation) {
69 share = 0;
70 }
71 a.qty += share;
72 left -= share;
73 }
74 }
75
76 // 3. Whatever is left goes FIFO, with no minimum. On a level of many small
77 // orders every share can round to zero, so this can be most of `qty`.
78 for (auto &a : allocations_) {
79 if (left == 0) {
80 break;
81 }
82 const qty_t extra = std::min(left, a.size - a.qty);
83 a.qty += extra;
84 left -= extra;
85 }
86 assert(left == 0);
87}
