46void TcpOrderGateway::onClientData(int client_id, const uint8_t *data, size_t length) {
47 auto it = partial_.find(client_id);
48 if (it == partial_.end() || it->second.empty()) {
49 // On a message boundary: parse straight out of the socket buffer, and keep
50 // only what the protocol could not use yet.
51 const size_t consumed = on_data(client_id, data, length);
52 if (consumed < length) {
53 partial_[client_id].assign(data + consumed, data + length);
54 }
55 return;
56 }
57
58 // Mid-message: the new bytes complete what is already buffered.
59 auto &buffer = it->second;
60 buffer.insert(buffer.end(), data, data + length);
61 const size_t consumed = on_data(client_id, buffer.data(), buffer.size());
62 // on_data may have closed the connection, which erases its entry only once this
63 // callback returns - so the iterator is still good here.
64 buffer.erase(buffer.begin(), buffer.begin() + static_cast<std::ptrdiff_t>(consumed));
65}
