122std::unique_ptr<MessageParser> TcpOrderGateway::create_parser_for_client(int client_id, const uint8_t* data, size_t length) {
123 ProtocolType protocol = default_protocol_;
124
125 // Check for client-specific protocol override
126 auto override_it = client_protocol_overrides_.find(client_id);
127 if (override_it != client_protocol_overrides_.end()) {
128 protocol = override_it->second;
129 LOG_DEBUG("Using protocol override for client {}: {}", client_id, static_cast<int>(protocol));
130 return MessageParserFactory::create_parser(protocol);
131 }
132
133 // Auto-detect protocol if enabled and data is available
134 if (auto_detect_protocol_ && data && length > 0) {
135 auto parser = MessageParserFactory::create_parser_from_data(data, length);
136 if (parser) {
137 LOG_DEBUG("Auto-detected protocol for client {}: {}",
138 client_id, static_cast<int>(parser->get_protocol_type()));
139 return parser;
140 }
141 }
142
143 // Fall back to default protocol
144 LOG_DEBUG("Using default protocol for client {}: {}", client_id, static_cast<int>(protocol));
145 return MessageParserFactory::create_parser(protocol);
146}
